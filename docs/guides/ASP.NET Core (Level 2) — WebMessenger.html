1. HTTP Pipeline

Этот раздел фокусируется на двух продвинутых темах конвейера: безопасном доступе к контексту запроса из произвольных сервисов через IHttpContextAccessor и динамическом создании фильтров с зависимостями через Filter Factory.

1.1 IHttpContextAccessor — доступ к контексту вне контроллера

Что это. IHttpContextAccessor — сервис, который хранит ссылку на текущий HttpContext в AsyncLocal<T>. Это позволяет любому сервису в DI-контейнере получить доступ к данным запроса (Claims, заголовки, IP) без явной передачи HttpContext через параметры методов.

Зачем. Контроллеры имеют прямой доступ к HttpContext через ControllerBase.HttpContext. Но сервисы, хелперы и инфраструктурные классы находятся вне контроллеров — им HttpContext напрямую недоступен. IHttpContextAccessor решает эту задачу.

Как работает. Middleware ASP.NET Core устанавливает HttpContextAccessor.HttpContext в начале каждого запроса и очищает в конце. Значение хранится в AsyncLocal<HttpContext> — это означает, что оно изолировано по асинхронному потоку выполнения и не пересекается между одновременными запросами.

Регистрация и использование в WebMessenger

// Program.cs — регистрация
builder.Services.AddHttpContextAccessor();
// ↑ Регистрирует IHttpContextAccessor как Singleton.
//   Singleton допустим потому что сам HttpContext хранится в AsyncLocal —
//   у каждого запроса своё изолированное значение, несмотря на Singleton-область сервиса.

builder.Services.AddScoped<ICurrentUser, CurrentUser>();
// ↑ CurrentUser — Scoped: новый экземпляр на каждый запрос.
//   Scoped гарантирует, что CurrentUser создаётся когда HttpContext уже установлен.

// Infrastructure/CurrentUser.cs — реальный код WebMessenger
public class CurrentUser(IHttpContextAccessor accessor) : ICurrentUser
{
    private readonly IHttpContextAccessor _accessor = accessor;

    public bool IsAuthenticated =>
        _accessor.HttpContext?.User?.Identity?.IsAuthenticated == true;
    //  ↑ Обратите внимание на ?. цепочку:
    //    HttpContext может быть null (фоновый сервис, вне запроса),
    //    User может быть null (незаполненный контекст),
    //    Identity может быть null (нет аутентификации).

    public Guid Id
    {
        get
        {
            if (!IsAuthenticated)
                throw new InvalidOperationException("User is not authenticated");
                // ↑ Явная ошибка вместо тихого возврата Guid.Empty —
                //   проще отладить, сложнее пропустить.

            var claim = _accessor.HttpContext!.User.FindFirst(ClaimTypes.NameIdentifier)
                        ?? throw new InvalidOperationException("NameIdentifier claim not found");
                        // ↑ ! оправдан: мы уже проверили IsAuthenticated выше,
                        //   поэтому HttpContext здесь точно не null.

            return Guid.Parse(claim.Value);
        }
    }

    public string? Username =>
        _accessor.HttpContext?.User?.FindFirst(ClaimTypes.Name)?.Value
        ?? _accessor.HttpContext?.User?.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        // ↑ Username необязателен (string?), поэтому null — допустимый результат.
}

// Пример использования ICurrentUser в сервисе (ChatService.cs)
public class ChatService(ICurrentUser currentUser, IUnitOfWork uow)
{
    public async Task<MessageDto> SendMessageAsync(SendMessageRequest request)
    {
        // Не нужно принимать userId как параметр — берём из контекста
        var senderId = currentUser.Id;

        // Это безопасно: если пользователь не аутентифицирован,
        // [Authorize] на контроллере уже вернул 401 до вызова этого метода.
        var message = new Message { SenderId = senderId, ... };
        await uow.MessageRepository.AddAsync(message);
    }
}

Ограничения IHttpContextAccessor

Ограничение Описание Решение
HttpContext может быть null Вне HTTP-запроса (Background Services, консольные задачи, тесты без хоста) HttpContext не установлен Всегда проверять ?.HttpContext != null перед использованием; не зависеть от IHttpContextAccessor в Background Services
Не работает корректно с await без AsyncLocal В старых версиях (IHttpContextAccessor на ThreadLocal) переключение потока могло потерять контекст. В ASP.NET Core используется AsyncLocal — проблемы нет В ASP.NET Core 2+ проблема решена. В классическом ASP.NET — актуально
Привязывает сервис к HTTP-запросу Сервис, зависящий от IHttpContextAccessor, нельзя переиспользовать вне HTTP-стека (например, в тестах или очередях) Извлекать нужные данные (userId, claims) в контроллере и передавать явно через параметры метода. ICurrentUser — слой абстракции, который это позволяет мокать в тестах
Singleton-сервис не может зависеть от Scoped-контекста Если зарегистрировать сервис с AddSingleton, DI не позволит внедрить Scoped-сервисы. Но IHttpContextAccessor сам является Singleton — это ОК Singleton может зависеть от IHttpContextAccessor (он Singleton), но не от самого HttpContext
Thread-safety при прямом изменении HttpContext Параллельная запись в HttpResponse из нескольких потоков не допускается Только чтение данных через IHttpContextAccessor; модификация Response — только в самом middleware/контроллере

Анти-паттерны использования IHttpContextAccessor

// ❌ АНТИ-ПАТТЕРН 1: Использование в Singleton-сервисе с прямым хранением HttpContext
public class BadSingleton : ISomeSingleton
{
    private readonly HttpContext _context; // НИКОГДА ТАК!

    public BadSingleton(IHttpContextAccessor accessor)
    {
        _context = accessor.HttpContext!; // ← захватываем ссылку на конкретный HttpContext
        // Singleton живёт до конца приложения.
        // Этот HttpContext будет завершён через ~100ms — утечка памяти и неопределённое поведение.
    }
}

// ✅ Правильно: всегда обращаться через accessor в момент вызова
public class GoodSingleton : ISomeSingleton
{
    private readonly IHttpContextAccessor _accessor; // храним accessor, не HttpContext

    public GoodSingleton(IHttpContextAccessor accessor) => _accessor = accessor;

    public string? GetCurrentUserId() =>
        _accessor.HttpContext?.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        // ← каждый раз получаем актуальный HttpContext текущего запроса
}

// ❌ АНТИ-ПАТТЕРН 2: Использование в Background Service
public class BadBackgroundService(IHttpContextAccessor accessor) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            // accessor.HttpContext здесь ВСЕГДА null — нет HTTP-запроса
            var userId = accessor.HttpContext?.User.FindFirst("sub")?.Value;
            // userId всегда null, логика тихо ломается
            await Task.Delay(1000, ct);
        }
    }
}

// ✅ Правильно: для Background Services данные берём из конфигурации,
// БД или передаём явно через Channel / Queue при публикации задачи
public record BackgroundJob(Guid UserId, string Payload);

public class GoodBackgroundService(Channel<BackgroundJob> channel) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        await foreach (var job in channel.Reader.ReadAllAsync(ct))
        {
            // userId пришёл явно от контроллера при постановке задачи в очередь
            await ProcessAsync(job.UserId, job.Payload, ct);
        }
    }
}

// ❌ АНТИ-ПАТТЕРН 3: Глубокая инъекция IHttpContextAccessor в доменный слой
// Domain/OrderService.cs — доменный сервис не должен знать об HTTP!
public class OrderDomainService(IHttpContextAccessor accessor)
{
    public void PlaceOrder(Order order)
    {
        // Доменный сервис не должен зависеть от HTTP-контекста —
        // это нарушает принцип разделения слоёв (Domain не зависит от Infrastructure)
        var userId = accessor.HttpContext!.User.FindFirst("sub")!.Value;
    }
}

// ✅ Правильно: Application Layer извлекает userId из HTTP,
// передаёт в Domain как обычный параметр
public class OrderAppService(IHttpContextAccessor accessor, OrderDomainService domain)
{
    public void PlaceOrder(OrderRequest request)
    {
        var userId = Guid.Parse(accessor.HttpContext!.User.FindFirst("sub")!.Value);
        domain.PlaceOrder(request.ToOrder(userId)); // чистый domain-вызов
    }
}
В WebMessenger паттерн корректен: ICurrentUser — это абстракция над IHttpContextAccessor, которая живёт в Infrastructure (не в Domain). В тестах можно легко подменить ICurrentUser моком без необходимости мокать IHttpContextAccessor.

1.2 Filter Factory

Что это. Filter Factory — механизм создания фильтров через DI-контейнер, а не напрямую через new. По умолчанию фильтры, применённые как атрибуты ([MyFilter]), создаются рефлексией — без участия DI, то есть без инъекции зависимостей в конструктор. Filter Factory решает эту проблему.

Зачем. Атрибут в C# — это обычный класс, но у атрибутов особые ограничения: они должны быть создаваемы без параметров или с константными значениями. Это значит, что у [MyFilter]-атрибута нельзя напрямую внедрить сервис через конструктор. Filter Factory обходит это, делегируя создание DI-контейнеру.

Три способа использовать фильтры с DI:

Подход Синтаксис DI? Когда использовать
Атрибут напрямую [MyFilter] Нет Фильтр без зависимостей (только параметры)
ServiceFilter [ServiceFilter(typeof(MyFilter))] Да Фильтр зарегистрирован в DI, простая инъекция
TypeFilter [TypeFilter(typeof(MyFilter), Arguments = new object[]{...})] Да Фильтр не зарегистрирован в DI, нужны runtime-аргументы
IFilterFactory Пользовательский атрибут реализует IFilterFactory Да Нужен контроль над созданием: условная логика, параметры + DI

ServiceFilter — фильтр из DI-контейнера

// Шаг 1: Фильтр с инъекцией зависимостей
public class AuditLogFilter(IAuditService auditService, ILogger<AuditLogFilter> logger)
    : IActionFilter
{
    public void OnActionExecuting(ActionExecutingContext context)
    {
        var user = context.HttpContext.User.FindFirst(ClaimTypes.Name)?.Value ?? "anonymous";
        var action = context.ActionDescriptor.DisplayName;
        logger.LogInformation("Action started: {Action} by {User}", action, user);
    }

    public void OnActionExecuted(ActionExecutedContext context)
    {
        auditService.LogAsync(context.ActionDescriptor.DisplayName, context.HttpContext);
    }
}

// Шаг 2: Обязательно зарегистрировать в DI (ServiceFilter требует регистрацию)
builder.Services.AddScoped<AuditLogFilter>();
// ↑ Scoped — у каждого запроса свой экземпляр фильтра.
//   Lifetime фильтра должен быть <= lifetime его зависимостей.

// Шаг 3: Применение через [ServiceFilter]
[ApiController]
[ServiceFilter(typeof(AuditLogFilter))]  // DI создаёт фильтр с его зависимостями
[Route("api/admin")]
public class AdminController : ControllerBase { }

// Или глобально для всех контроллеров:
builder.Services.AddControllers(options =>
{
    options.Filters.AddService<AuditLogFilter>();
    // ↑ AddService<T>() аналогичен [ServiceFilter(typeof(T))] но для глобального применения
});

TypeFilter — фильтр с runtime-параметрами и DI

// TypeFilter создаёт экземпляр фильтра через Microsoft.Extensions.DependencyInjection.ObjectFactory,
// передавая дополнительные аргументы и разрешая зависимости из DI.
// Регистрировать фильтр в DI НЕ нужно.

public class RequireRoleFilter(string role, ILogger<RequireRoleFilter> logger)
    : IAuthorizationFilter
{
    public void OnAuthorization(AuthorizationFilterContext context)
    {
        if (!context.HttpContext.User.IsInRole(role))
        {
            logger.LogWarning("Access denied. Required role: {Role}", role);
            context.Result = new ForbidResult();
        }
    }
}

// Применение — role передаётся как runtime-аргумент:
[TypeFilter(typeof(RequireRoleFilter), Arguments = new object[] { "Admin" })]
[HttpDelete("users/{id}")]
public IActionResult DeleteUser(Guid id) { ... }

// ↑ TypeFilter создаёт RequireRoleFilter(role: "Admin", logger: <из DI>)
//   — аргументы Arguments комбинируются с зависимостями из DI

IFilterFactory — полный контроль над созданием фильтра

// IFilterFactory — интерфейс, который позволяет атрибуту самому управлять
// созданием фильтра. Это даёт максимальную гибкость: сочетание параметров
// атрибута с зависимостями из DI, условная логика создания.

// Шаг 1: Атрибут реализует IFilterFactory
[AttributeUsage(AttributeTargets.Method | AttributeTargets.Class)]
public class RateLimitAttribute(int maxRequests, int windowSeconds) : Attribute, IFilterFactory
{
    // IsReusable = true  → один экземпляр фильтра для всего приложения (Singleton-like)
    // IsReusable = false → новый экземпляр при каждом запросе (Transient-like)
    // Выбирайте false, если фильтр хранит состояние запроса.
    public bool IsReusable => false;

    public IFilterMetadata CreateInstance(IServiceProvider serviceProvider)
    {
        // Получаем зависимости из DI напрямую
        var cache = serviceProvider.GetRequiredService<IDistributedCache>();
        var logger = serviceProvider.GetRequiredService<ILogger<RateLimitFilter>>();

        // Создаём фильтр, комбинируя параметры атрибута с DI-зависимостями
        return new RateLimitFilter(maxRequests, windowSeconds, cache, logger);
    }
}

// Шаг 2: Сам фильтр — не атрибут, обычный класс
public class RateLimitFilter(
    int maxRequests,
    int windowSeconds,
    IDistributedCache cache,
    ILogger<RateLimitFilter> logger) : IAsyncActionFilter
{
    public async Task OnActionExecutionAsync(
        ActionExecutingContext context,
        ActionExecutionDelegate next)
    {
        var ip = context.HttpContext.Connection.RemoteIpAddress?.ToString() ?? "unknown";
        var key = $"rl:{ip}:{context.ActionDescriptor.DisplayName}";

        var countBytes = await cache.GetAsync(key);
        var count = countBytes is null ? 0 : BitConverter.ToInt32(countBytes);

        if (count >= maxRequests)
        {
            logger.LogWarning("Rate limit exceeded for IP {Ip}", ip);
            context.Result = new StatusCodeResult(429); // Too Many Requests
            return;
        }

        await cache.SetAsync(key,
            BitConverter.GetBytes(count + 1),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromSeconds(windowSeconds)
            });

        await next(); // продолжить выполнение action
    }
}

// Шаг 3: Применение — выглядит как обычный атрибут
[RateLimit(maxRequests: 10, windowSeconds: 60)]
[HttpPost("login")]
public IActionResult Login([FromBody] LoginDto dto) { ... }

// Параметры атрибута сохраняются и передаются в CreateInstance — чисто и читаемо.

Сравнение: ServiceFilter vs TypeFilter vs IFilterFactory

// ServiceFilter — подходит для простых случаев, фильтр уже в DI
[ServiceFilter(typeof(AuditLogFilter))]
public IActionResult Get() { ... }

// TypeFilter — подходит когда нужны runtime-аргументы, но не хочется писать атрибут
[TypeFilter(typeof(RequireRoleFilter), Arguments = new object[] { "Admin" })]
public IActionResult AdminAction() { ... }

// IFilterFactory — максимальная гибкость и читаемость использования
// выглядит как "чистый" атрибут с параметрами, при этом полностью поддерживает DI
[RateLimit(maxRequests: 5, windowSeconds: 30)]
public IActionResult SensitiveAction() { ... }

// Выбор:
// - Нет параметров, фильтр в DI        → ServiceFilter
// - Есть параметры, фильтр не в DI     → TypeFilter
// - Есть параметры + DI + чистый API   → IFilterFactory (лучший вариант)
IsReusable и состояние. Если установить IsReusable = true, ASP.NET Core кэширует экземпляр фильтра и переиспользует его между запросами. Это опасно если фильтр хранит состояние (поля, изменяемые в ходе запроса). При IsReusable = true фильтр должен быть полностью stateless — как Singleton.
В WebMessenger фильтры не используются напрямую — аутентификация реализована через стандартный [Authorize], а аудит отсутствует. Filter Factory — хорошая точка расширения для добавления Rate Limiting или Audit-логирования без изменения контроллеров.

2. Routing

Продвинутые механизмы маршрутизации: генерация URL, тонкая настройка ограничений маршрутов, метаданные эндпоинтов и обработка запросов, не попавших ни в один маршрут.

2.1 Link Generation — генерация URL

Что это. Link Generation — механизм построения URL по имени маршрута или типу контроллера/action вместо ручной конкатенации строк. При изменении шаблона маршрута генерируемые URL обновляются автоматически.

Зачем. Жёстко заданные строки URL ломаются при переименовании action, изменении prefix или переезде на версионирование. Link Generation делает URL compile-time зависимостью, а не runtime-константой.

Три инструмента генерации URL

Инструмент Доступен Применение
IUrlHelper (Url) В контроллерах (через ControllerBase.Url) Генерация URL из action-методов контроллера. Учитывает текущий Request
LinkGenerator Везде через DI Генерация URL из любого сервиса/middleware. Не требует HttpContext
CreatedAtAction / CreatedAtRoute В контроллерах Ответ 201 Created с заголовком Location на созданный ресурс

IUrlHelper — генерация из контроллера

// IUrlHelper доступен в контроллерах через унаследованное свойство Url
[ApiController]
[Route("api/users")]
public class UserController : ControllerBase
{
    [HttpPost]
    public IActionResult CreateUser([FromBody] CreateUserRequest request)
    {
        var user = _userService.Create(request);

        // Url.Action() генерирует относительный URL по имени action и контроллера
        var url = Url.Action(
            action: nameof(GetUserProfileById), // "GetUserProfileById"
            controller: "User",                 // имя контроллера без суффикса "Controller"
            values: new { id = user.Id },       // значения параметров маршрута
            protocol: Request.Scheme            // "http" или "https" — делает URL абсолютным
        );
        // url = "https://localhost:5000/api/users/profile/3fa85f64-5717-4562-b3fc-2c963f66afa6"

        return Created(url!, user);
        // Эквивалентно CreatedAtAction, но URL уже готов
    }

    // Именованный маршрут — для использования с Url.RouteUrl()
    [HttpGet("profile/{id:guid}", Name = "GetUserProfile")]
    public async Task GetUserProfileById(Guid id) { ... }

    [HttpPost("invite")]
    public IActionResult GenerateInviteLink(Guid targetUserId)
    {
        // Url.RouteUrl() — генерация по имени маршрута (Name = "...")
        var profileUrl = Url.RouteUrl(
            routeName: "GetUserProfile",
            values: new { id = targetUserId },
            protocol: Request.Scheme,
            host: Request.Host.Value
        );
        return Ok(new { inviteUrl = profileUrl });
    }
}

LinkGenerator — генерация вне контроллера

// LinkGenerator — сервис, который работает без HttpContext.
// Полезен в: Background Services, Email-сервисах, Middleware, обработчиках событий.

public class NotificationService(LinkGenerator linkGenerator)
{
    public string BuildPasswordResetLink(Guid userId, string token, string scheme, HostString host)
    {
        // GetUriByAction — генерирует абсолютный URI (со схемой и хостом)
        var uri = linkGenerator.GetUriByAction(
            action: "ResetPassword",     // имя action-метода
            controller: "Auth",          // имя контроллера (без суффикса)
            values: new { userId, token }, // параметры маршрута / query string
            scheme: scheme,              // "https"
            host: host                   // "app.example.com"
        );
        // uri = "https://app.example.com/api/auth/reset-password?userId=...&token=..."

        return uri ?? throw new InvalidOperationException("Failed to generate link");
    }

    public string BuildRelativeProfileUrl(Guid userId)
    {
        // GetPathByAction — генерирует только путь (без схемы и хоста)
        var path = linkGenerator.GetPathByAction(
            action: "GetUserProfileById",
            controller: "User",
            values: new { id = userId }
        );
        // path = "/api/users/profile/3fa85f64-5717-4562-b3fc-2c963f66afa6"

        return path ?? throw new InvalidOperationException("Failed to generate path");
    }
}

CreatedAtAction — ответ 201 с Location

// REST-конвенция: при создании ресурса возвращать 201 Created
// с заголовком Location: URL нового ресурса.
// CreatedAtAction делает это в одну строку.

[ApiController]
[Route("api/chats")]
public class ChatController : ControllerBase
{
    [HttpPost]
    public async Task CreateChat([FromBody] CreateChatRequest request)
    {
        var chat = await _chatService.CreateChatAsync(request);

        // CreatedAtAction генерирует Location header и возвращает 201:
        return CreatedAtAction(
            actionName: nameof(GetChatHeader),  // action для получения созданного ресурса
            routeValues: new { chatId = chat.Id }, // параметры этого action
            value: chat                         // тело ответа
        );
        // Location: https://localhost:5000/api/chats/3fa85f64.../header
        // Status:   201 Created
        // Body:     { "id": "3fa85f64...", ... }
    }

    [HttpGet("{chatId:guid}/header")]
    public async Task GetChatHeader(Guid chatId) { ... }
}

// CreatedAtRoute — когда у action есть именованный маршрут:
[HttpPost]
public IActionResult CreateUser([FromBody] CreateUserDto dto)
{
    var user = _service.Create(dto);
    return CreatedAtRoute(
        routeName: "GetUserProfile", // [HttpGet("profile/{id:guid}", Name = "GetUserProfile")]
        routeValues: new { id = user.Id },
        value: user
    );
}

2.2 Route Constraints — ограничения маршрутов

Что это. Route Constraints — условия, которым должны удовлетворять значения сегментов маршрута при сопоставлении. Без ограничения любой сегмент URL будет соответствовать параметру. Ограничения позволяют маршрутизатору различать /api/users/42 (int) и /api/users/profile (string).

Важно. Constraints — это инструмент разрешения маршрутов, а не валидации входных данных. Если значение не удовлетворяет ограничению, маршрут просто не выбирается (404), а не возвращается 400. Для бизнес-валидации используйте Data Annotations.

[HttpGet("{chatId:guid}/header")]
// ↑ Соответствует: GET /api/chats/3fa85f64-5717-4562-b3fc-2c963f66afa6/header
// Не соответствует: GET /api/chats/123/header  → следующий подходящий маршрут или 404

// Без :guid маршруты "direct/{userId}/header" и "{chatId}/header" могли бы конфликтовать.
// :guid разделяет их: GUID в первом сегменте → первый маршрут,
// строка "direct" → второй маршрут.

[HttpGet("profile/{id:guid}")]
// Не конфликтует с [HttpGet("profile")] —
// при запросе GET /api/users/profile маршрут с :guid не сработает,
// сработает маршрут без параметра.

// Несколько ограничений — через двоеточие:
[HttpGet("{id:guid:required}")]

2.3 Custom Route Constraints via IRouteConstraint

Что это. IRouteConstraint — интерфейс для создания собственных ограничений маршрутов. Если встроенных constraint'ов недостаточно (например, нужно проверить, что значение соответствует определённому бизнес-правилу или формату), можно реализовать собственный.

// Пример: ограничение на формат имени пользователя — только строчные буквы и цифры
public class UsernameRouteConstraint : IRouteConstraint
{
    // Регулярное выражение для допустимых имён пользователей WebMessenger
    private static readonly Regex _usernamePattern =
        new(@"^[a-z0-9_]{3,50}$", RegexOptions.Compiled | RegexOptions.CultureInvariant);

    public bool Match(
        HttpContext? httpContext,  // может быть null при генерации URL (link generation)
        IRouter? route,           // устаревший параметр, обычно null в endpoint routing
        string routeKey,          // имя параметра маршрута, например "username"
        RouteValueDictionary values, // текущие значения маршрута
        RouteDirection routeDirection) // IncomingRequest или UrlGeneration
    {
        // Получаем значение параметра
        if (!values.TryGetValue(routeKey, out var value) || value is null)
            return false;

        var stringValue = value.ToString()!;

        // При генерации URL (link generation) — проверяем формат, чтобы не генерировать
        // невалидные URL через LinkGenerator
        return _usernamePattern.IsMatch(stringValue);
    }
}

// Регистрация пользовательского constraint в Program.cs
builder.Services.Configure<RouteOptions>(options =>
{
    options.ConstraintMap.Add("username", typeof(UsernameRouteConstraint));
    // Ключ "username" теперь можно использовать в шаблонах маршрутов
});

// Применение в контроллере
[HttpGet("{username:username}")]
// ↑ Соответствует: GET /api/users/alice_1  (строчные буквы, цифры, _, 3-50 символов)
//   Не соответствует: GET /api/users/Alice (большая буква)
//   Не соответствует: GET /api/users/ab   (слишком короткое)
public async Task GetByUsername(string username) { ... }

// Пример 2: ограничение на версию API в URL (подготовка к версионированию)
public class ApiVersionConstraint(string allowedVersion) : IRouteConstraint
{
    public bool Match(
        HttpContext? httpContext,
        IRouter? route,
        string routeKey,
        RouteValueDictionary values,
        RouteDirection routeDirection)
    {
        if (!values.TryGetValue(routeKey, out var value)) return false;

        // Допускаем только конкретные форматы: "v1", "v2", "v1.1"
        var version = value?.ToString();
        return version == allowedVersion;
    }
}

// Регистрация:
options.ConstraintMap.Add("apiversion", typeof(ApiVersionConstraint));

// Это нестандартный подход — см. Раздел 5 для правильного API Versioning.
// Constraint с параметром не работает через ConstraintMap без фабрики.
// Корректный способ — стандартные constraints или Asp.Versioning пакет.
Ограничения Custom Constraints: Если constraint требует конструктор с параметрами (например, allowedVersion), то ConstraintMap не может его разрешить напрямую — нужно либо реализовать IInlineConstraintResolver, либо использовать более подходящий подход (именованные маршруты, версионирование через атрибуты).

2.4 Route Endpoint Metadata

Что это. Metadata эндпоинта — набор объектов, прикреплённых к маршруту, которые описывают его свойства. Middleware и фильтры могут читать эту metadata для принятия решений — например, требует ли эндпоинт аутентификации, какие политики авторизации применяются, нужен ли rate limiting.

Как работает. Атрибуты на контроллерах и action'ах ([Authorize], [AllowAnonymous], [HttpGet], [Produces]) — это и есть metadata. ASP.NET Core собирает их при сопоставлении маршрута и хранит в Endpoint.Metadata.

Чтение metadata в middleware

// Пример: middleware, который проверяет наличие кастомного атрибута-metadata
// и логирует обращения к "чувствительным" эндпоинтам

// Шаг 1: Создаём маркерный атрибут — он же будет metadata-объектом
[AttributeUsage(AttributeTargets.Method | AttributeTargets.Class)]
public class SensitiveEndpointAttribute : Attribute
{
    public string? Reason { get; init; }
}

// Шаг 2: Применяем атрибут к action
[HttpDelete("{chatId:guid}")]
[SensitiveEndpoint(Reason = "Deletes all chat history")]
public async Task DeleteChat(Guid chatId) { ... }

// Шаг 3: Читаем metadata в middleware
public class SensitiveEndpointAuditMiddleware(RequestDelegate next, IAuditService audit)
{
    public async Task InvokeAsync(HttpContext context)
    {
        // IEndpointFeature установлен ПОСЛЕ UseRouting() — до него endpoint ещё не определён
        var endpoint = context.GetEndpoint();

        if (endpoint is not null)
        {
            var metadata = endpoint.Metadata.GetMetadata<SensitiveEndpointAttribute>();
            if (metadata is not null)
            {
                var user = context.User.FindFirst(ClaimTypes.Name)?.Value ?? "anonymous";
                await audit.LogSensitiveAccessAsync(
                    user,
                    endpoint.DisplayName,
                    metadata.Reason
                );
            }
        }

        await next(context);
    }
}

// Встроенная metadata — примеры из самого ASP.NET Core:

// context.GetEndpoint()?.Metadata.GetMetadata<IAuthorizeData>()
// → не null означает, что эндпоинт требует авторизации ([Authorize])

// context.GetEndpoint()?.Metadata.GetMetadata<IAllowAnonymous>()
// → не null означает, что анонимный доступ явно разрешён ([AllowAnonymous])

// context.GetEndpoint()?.Metadata.GetMetadata<HttpMethodMetadata>()?.HttpMethods
// → список HTTP-методов: ["GET"], ["POST"], ["GET", "HEAD"]

// context.GetEndpoint()?.Metadata.GetMetadata<RouteNameMetadata>()?.RouteName
// → имя маршрута если задано через Name = "..."

// context.GetEndpoint()?.DisplayName
// → читаемое имя: "WebMessenger.Api.Controllers.ChatController.GetChatHeader (WebMessenger.Api)"

Добавление metadata программно через MapGet/MapPost

// При использовании Minimal API или кастомных эндпоинтов:
app.MapGet("/api/health", () => Results.Ok(new { status = "healthy" }))
    .WithMetadata(new SensitiveEndpointAttribute { Reason = "Internal health check" })
    .WithName("HealthCheck")         // RouteNameMetadata
    .WithTags("Infrastructure")      // OpenAPI тег
    .RequireAuthorization("AdminOnly"); // IAuthorizeData

// Для контроллеров metadata добавляется через атрибуты — это единственный способ.
// Программная установка metadata для контроллеров через IEndpointConventionBuilder
// требует использования конвенций (IControllerModelConvention).

2.5 Fallback Routing

Что это. Fallback Routing — специальный маршрут с наименьшим приоритетом, который соответствует любому запросу, который не совпал с другими маршрутами. Это альтернатива 404 «по умолчанию» — можно вернуть осмысленный ответ или перенаправить.

Зачем. Типичные применения:

  • SPA (Single Page Application) — все «фронтендовые» пути (/profile, /chats/42) должны вернуть index.html чтобы клиентский роутер мог обработать путь
  • Кастомная страница 404 вместо стандартного ответа
  • API Gateway — пересылка неизвестных путей в upstream-сервис
  • Логирование несуществующих маршрутов

MapFallback — конфигурация Fallback

// MapFallback срабатывает для ЛЮБОГО метода и ЛЮБОГО пути,
// который не совпал ни с одним другим маршрутом.

// Вариант 1: Fallback с inline-обработчиком
app.MapFallback(async context =>
{
    context.Response.StatusCode = 404;
    context.Response.ContentType = "application/json";
    await context.Response.WriteAsJsonAsync(new
    {
        status = 404,
        message = $"Endpoint '{context.Request.Path}' not found",
        hint   = "See /swagger for available endpoints"
    });
});

// Вариант 2: Fallback для SPA — отдать index.html
// (WebMessenger использует Next.js на отдельном порту, поэтому здесь не нужно)
app.MapFallbackToFile("index.html");
// ↑ Ищет index.html в wwwroot и возвращает его с 200.
//   Позволяет React/Vue/Next.js обработать путь на клиенте.

// Вариант 3: Fallback на контроллер
app.MapFallbackToController("NotFound", "Error");
// GET /any-unknown-path → ErrorController.NotFound()

// Вариант 4: Fallback только для конкретного паттерна
app.MapFallback("/api/{**slug}", async context =>
{
    // Срабатывает только для путей начинающихся с /api/,
    // которые не соответствуют ни одному контроллеру
    context.Response.StatusCode = 404;
    await context.Response.WriteAsJsonAsync(new ProblemDetails
    {
        Status = 404,
        Title  = "API Endpoint Not Found",
        Detail = $"No API endpoint found at {context.Request.Method} {context.Request.Path}"
    });
});
// Общий MapFallback без паттерна обрабатывает всё остальное.

Порядок приоритетов маршрутов

// ASP.NET Core выбирает маршрут по приоритету (от наивысшего к наименьшему):

// 1. Точное совпадение литерала
app.MapGet("/api/users/me", ...);          // → GET /api/users/me

// 2. Сегменты с ограничениями
app.MapControllers();                       // → GET /api/users/profile/{id:guid}

// 3. Сегменты без ограничений
app.MapGet("/api/{**path}", ...);          // → любой путь под /api/

// 4. MapFallback — всегда самый низкий приоритет
app.MapFallback(...);                       // → всё остальное

// Важно: MapFallback должен регистрироваться ПОСЛЕ всех остальных маршрутов.
// Порядок вызовов app.Map* определяет порядок проверки при одинаковом приоритете.

// Catch-all параметр {**rest} vs Fallback:
// {**rest} — это параметр маршрута, он всё ещё часть конкретного шаблона
// MapFallback — ниже любого {**rest} шаблона

Fallback в контексте WebMessenger

// WebMessenger Program.cs — маршруты регистрируются так:
app.MapControllers();
// → все [Route("api/...")] контроллеры

app.MapHub<ChatHub>("/hubs/chat").RequireCors(FrontCors);
// → WebSocket-соединение к SignalR-хабу

// Запросы к несуществующим эндпоинтам сейчас возвращают стандартный ProblemDetails 404
// (благодаря app.UseExceptionHandler() + AddProblemDetails()).

// Рекомендация: добавить Fallback для явного ответа на неизвестные /api/* пути:
app.MapFallback("/api/{**slug}", async context =>
{
    context.Response.StatusCode = 404;
    await context.Response.WriteAsJsonAsync(new ProblemDetails
    {
        Status  = 404,
        Title   = "Not Found",
        Detail  = $"No endpoint matched {context.Request.Method} {context.Request.Path}",
        Extensions = { ["traceId"] = context.TraceIdentifier }
    });
});
MapFallbackToFile vs StaticFiles. MapFallbackToFile работает только если UseStaticFiles() включён и файл существует в wwwroot. В WebMessenger фронтенд (Next.js) работает на порту 3000 и является отдельным процессом — статические файлы из wwwroot не используются для SPA-маршрутизации.

3. Validation

Этот раздел охватывает продвинутые сценарии валидации: создание собственных атрибутов, межполевую валидацию, вложенные объекты и тонкую настройку ProblemDetails-ответа при ошибках валидации.

3.1 Custom Data Annotations — ValidationAttribute

Что это. ValidationAttribute — базовый класс всех атрибутов валидации в .NET ([Required], [MaxLength] и т.д.). Наследуясь от него, можно создать собственный атрибут с произвольной логикой проверки, который работает везде, где работают стандартные атрибуты: в MVC, при ручном вызове Validator.TryValidateObject, в Swagger.

Зачем. Стандартных атрибутов не хватает для бизнес-правил: «имя пользователя не должно содержать мат», «дата начала должна быть в будущем», «значение должно быть из списка допустимых». Custom ValidationAttribute позволяет инкапсулировать такие правила в переиспользуемые атрибуты.

Базовый Custom ValidationAttribute

// Атрибут: имя пользователя — только буквы, цифры и подчёркивание
[AttributeUsage(AttributeTargets.Property | AttributeTargets.Field | AttributeTargets.Parameter)]
public class UsernameAttribute : ValidationAttribute
{
    // Regex компилируется один раз — атрибут может быть применён к множеству свойств
    private static readonly Regex _pattern =
        new(@"^[a-zA-Z0-9_]+$", RegexOptions.Compiled | RegexOptions.CultureInvariant);

    public UsernameAttribute()
    {
        // ErrorMessage — сообщение по умолчанию; можно переопределить при использовании атрибута
        ErrorMessage = "Username may only contain letters, digits and underscores.";
    }

    // Основной метод — вся логика проверки здесь
    protected override ValidationResult? IsValid(object? value, ValidationContext validationContext)
    {
        // null считаем валидным — за обязательность отвечает [Required]
        if (value is null) return ValidationResult.Success;

        var username = value.ToString()!;

        if (!_pattern.IsMatch(username))
        {
            // ValidationResult принимает сообщение и список "виновных" полей
            return new ValidationResult(
                ErrorMessage,
                [validationContext.MemberName!]
                // ↑ MemberName = "Username" — имя свойства на котором применён атрибут.
                //   Важно передавать: ModelState группирует ошибки по имени поля.
            );
        }

        return ValidationResult.Success;
    }
}

// Применение в DTO:
public sealed record RegisterDto
{
    [Required]
    [MaxLength(50)]
    [Username]                    // ← наш кастомный атрибут
    public required string Username { get; init; }

    [Required]
    [MinLength(6)]
    [MaxLength(100)]
    public required string Password { get; init; }
}

ValidationAttribute с параметрами

// Атрибут: допустимые значения из заданного набора (enum-like для строк)
public class AllowedValuesAttribute(params string[] allowedValues) : ValidationAttribute
{
    private readonly HashSet<string> _allowed =
        new(allowedValues, StringComparer.OrdinalIgnoreCase);

    protected override ValidationResult? IsValid(object? value, ValidationContext ctx)
    {
        if (value is null) return ValidationResult.Success;

        var str = value.ToString()!;
        if (!_allowed.Contains(str))
        {
            return new ValidationResult(
                $"Value '{str}' is not allowed. Allowed values: {string.Join(", ", allowedValues)}",
                [ctx.MemberName!]
            );
        }

        return ValidationResult.Success;
    }
}

// Применение:
public record UpdateStatusRequest
{
    [Required]
    [AllowedValues("online", "offline", "away", "busy")]
    public required string Status { get; init; }
}

ValidationAttribute с доступом к сервисам через ValidationContext

// ValidationContext.GetService<T>() позволяет получить DI-сервисы
// ВАЖНО: это работает только при вызове через MVC (HTTP-запрос),
// но НЕ при ручном Validator.TryValidateObject() без настроенного ServiceProvider.

public class UniqueUsernameAttribute : ValidationAttribute
{
    protected override ValidationResult? IsValid(object? value, ValidationContext ctx)
    {
        if (value is null) return ValidationResult.Success;

        // Получаем сервис из DI через ValidationContext
        var userRepo = ctx.GetService<IUserRepository>();
        if (userRepo is null)
            return ValidationResult.Success; // в unit-тестах сервис может отсутствовать

        var username = value.ToString()!;
        var exists = userRepo.ExistsByUsernameAsync(username).GetAwaiter().GetResult();
        // ↑ GetAwaiter().GetResult() — вынужденное решение: IsValid синхронный.
        //   Для async-валидации лучше использовать FluentValidation или ручную проверку в сервисе.

        if (exists)
        {
            return new ValidationResult(
                $"Username '{username}' is already taken.",
                [ctx.MemberName!]
            );
        }

        return ValidationResult.Success;
    }
}
Async в ValidationAttribute. Метод IsValid синхронный. Блокировка через .GetAwaiter().GetResult() внутри async-контекста может привести к deadlock в некоторых хостинговых средах. Для async-валидации с доступом к БД рекомендуется FluentValidation или явная проверка в сервисном слое перед сохранением.

3.2 Cross-Property Validation

Что это. Cross-property (межполевая) валидация — проверка, которая зависит от значений нескольких свойств одновременно. Например: «дата окончания должна быть позже даты начала», «если поле A заполнено, то поле B обязательно».

Проблема стандартного подхода. Атрибут на свойстве видит только значение этого свойства. Чтобы сравнить с другим свойством, нужен доступ ко всему объекту — и для этого есть два пути: ValidationAttribute на уровне класса или IValidatableObject.

ValidationAttribute на уровне класса

// Атрибут применяется ко всему классу (AttributeTargets.Class)
// и получает весь объект в value
[AttributeUsage(AttributeTargets.Class)]
public class DateRangeAttribute : ValidationAttribute
{
    private readonly string _startPropName;
    private readonly string _endPropName;

    public DateRangeAttribute(string startPropertyName, string endPropertyName)
    {
        _startPropName = startPropertyName;
        _endPropName   = endPropertyName;
        ErrorMessage   = $"{endPropertyName} must be after {startPropertyName}.";
    }

    protected override ValidationResult? IsValid(object? value, ValidationContext ctx)
    {
        if (value is null) return ValidationResult.Success;

        var type  = value.GetType();
        var start = (DateTime?)type.GetProperty(_startPropName)?.GetValue(value);
        var end   = (DateTime?)type.GetProperty(_endPropName)?.GetValue(value);

        if (start.HasValue && end.HasValue && end.Value <= start.Value)
        {
            return new ValidationResult(
                ErrorMessage,
                [_endPropName]  // ошибка привязана к полю EndDate
            );
        }

        return ValidationResult.Success;
    }
}

// Применение на уровне класса:
[DateRange(nameof(StartDate), nameof(EndDate))]
public record CreateEventRequest
{
    [Required]
    public DateTime StartDate { get; init; }

    [Required]
    public DateTime EndDate { get; init; }

    [Required]
    [MaxLength(200)]
    public required string Title { get; init; }
}

CompareAttribute — встроенная межполевая валидация

// [Compare] — встроенный атрибут для проверки равенства двух полей
// Типичный сценарий: подтверждение пароля при регистрации
public sealed record RegisterWithConfirmDto
{
    [Required]
    [MinLength(6)]
    [MaxLength(100)]
    public required string Password { get; init; }

    [Required]
    [Compare(nameof(Password), ErrorMessage = "Passwords do not match.")]
    // ↑ [Compare] сравнивает значение этого свойства со значением Password
    public required string ConfirmPassword { get; init; }
}

3.3 IValidatableObject — модельная валидация

Что это. IValidatableObject — интерфейс с единственным методом Validate(ValidationContext). Реализовав его на DTO/модели, можно добавить произвольную логику валидации, которая выполняется после всех атрибутов (если атрибуты прошли успешно).

Когда использовать вместо класс-атрибута.

  • Логика специфична для этого конкретного класса и нигде не переиспользуется
  • Нужна сложная многоэтапная проверка с несколькими ошибками одновременно
  • Логика требует доступа к вычисляемым свойствам или методам класса
public sealed record ScheduleMessageRequest : IValidatableObject
{
    [Required]
    [MaxLength(5000)]
    public required string Content { get; init; }

    public DateTime? ScheduledAt { get; init; }

    public bool IsRecurring { get; init; }

    [Range(1, 365)]
    public int? RecurrenceIntervalDays { get; init; }

    // Validate() вызывается ТОЛЬКО если все атрибуты прошли успешно.
    // Возвращает IEnumerable<ValidationResult> — можно вернуть несколько ошибок.
    public IEnumerable<ValidationResult> Validate(ValidationContext validationContext)
    {
        // Правило 1: ScheduledAt должен быть в будущем
        if (ScheduledAt.HasValue && ScheduledAt.Value <= DateTime.UtcNow)
        {
            yield return new ValidationResult(
                "ScheduledAt must be in the future.",
                [nameof(ScheduledAt)]
            );
        }

        // Правило 2: IsRecurring требует оба: ScheduledAt и RecurrenceIntervalDays
        if (IsRecurring)
        {
            if (!ScheduledAt.HasValue)
                yield return new ValidationResult(
                    "ScheduledAt is required for recurring messages.",
                    [nameof(ScheduledAt)]
                );

            if (!RecurrenceIntervalDays.HasValue)
                yield return new ValidationResult(
                    "RecurrenceIntervalDays is required for recurring messages.",
                    [nameof(RecurrenceIntervalDays)]
                );
        }

        // Если ошибок нет — метод просто заканчивается (yield не вызван).
        // yield break; — явное завершение, необязательно
    }
}

// Порядок выполнения валидации в ASP.NET Core:
// 1. Проверяется наличие required-свойств (C# required keyword / [Required])
// 2. Выполняются ValidationAttribute на свойствах (MaxLength, MinLength, etc.)
// 3. Выполняются ValidationAttribute на классе
// 4. Если шаги 1-3 прошли без ошибок → вызывается IValidatableObject.Validate()
//
// Важно: если атрибуты нашли ошибки, Validate() НЕ вызывается.
// Это "fail-fast" поведение — сначала проверяем простые ограничения.

3.5 Custom ProblemDetails — настройка ответа об ошибке валидации

Что это. Когда [ApiController] обнаруживает невалидный ModelState, он автоматически возвращает 400 Bad Request в формате ValidationProblemDetails (расширение RFC 7807). Форму этого ответа можно полностью контролировать через ConfigureApiBehaviorOptions.

Структура стандартного ValidationProblemDetails

// Что возвращает [ApiController] по умолчанию при ошибке валидации:
{
  "type": "https://tools.ietf.org/html/rfc9110#section-15.5.1",
  "title": "One or more validation errors occurred.",
  "status": 400,
  "errors": {
    "Username": ["The Username field is required.", "Username may only contain letters..."],
    "Password": ["The field Password must have a minimum length of 6."]
  },
  "traceId": "00-abc123-def456-00"
}
// "errors" — словарь: ключ = имя поля, значение = массив сообщений об ошибках

Настройка через ConfigureApiBehaviorOptions

// Program.cs — полная настройка ответа об ошибке валидации
builder.Services.AddControllers()
    .ConfigureApiBehaviorOptions(options =>
    {
        // InvalidModelStateResponseFactory — фабрика, создающая ответ при невалидном ModelState.
        // Заменяет поведение по умолчанию полностью.
        options.InvalidModelStateResponseFactory = context =>
        {
            // Собираем все ошибки из ModelState в плоский список
            var errors = context.ModelState
                .Where(e => e.Value?.Errors.Count > 0)
                .SelectMany(kvp => kvp.Value!.Errors.Select(e => new
                {
                    field   = kvp.Key,     // "Username", "Password", "Address.ZipCode"
                    message = e.ErrorMessage
                }))
                .ToList();

            // Создаём расширенный ProblemDetails
            var problemDetails = new ValidationProblemDetails(context.ModelState)
            {
                Type    = "https://tools.ietf.org/html/rfc9110#section-15.5.1",
                Title   = "Validation Failed",
                Status  = StatusCodes.Status400BadRequest,
                Detail  = $"{errors.Count} validation error(s) occurred.",
                Instance = context.HttpContext.Request.Path
                // Instance — URI конкретного запроса, который вызвал ошибку
            };

            // Добавляем кастомные расширения
            problemDetails.Extensions["traceId"]   = context.HttpContext.TraceIdentifier;
            problemDetails.Extensions["timestamp"]  = DateTimeOffset.UtcNow;
            problemDetails.Extensions["requestId"]  = context.HttpContext.Request.Headers
                                                              .TryGetValue("X-Request-Id", out var rid)
                                                              ? (string)rid!
                                                              : null;

            return new BadRequestObjectResult(problemDetails)
            {
                ContentTypes = { "application/problem+json" }
                // ↑ Правильный Content-Type для RFC 7807 ProblemDetails
            };
        };
    });

Ситуация в WebMessenger и что улучшить

// WebMessenger — текущее состояние:
builder.Services.AddControllers();
// ↑ Без .ConfigureApiBehaviorOptions() — ответ об ошибке валидации стандартный.
// AddProblemDetails() добавляет только traceId через CustomizeProblemDetails,
// но это не применяется к ValidationProblemDetails из [ApiController].

// Текущий ответ при невалидном RegisterDto:
// { "type": "...", "title": "One or more validation errors occurred.", "status": 400,
//   "errors": { "Username": [...], "Password": [...] } }
// traceId НЕ добавляется (AddProblemDetails CustomizeProblemDetails не вызывается для 400)

// Рекомендация: добавить ConfigureApiBehaviorOptions с traceId:
builder.Services.AddControllers()
    .ConfigureApiBehaviorOptions(options =>
    {
        options.InvalidModelStateResponseFactory = context =>
        {
            var pd = new ValidationProblemDetails(context.ModelState)
            {
                Status   = 400,
                Instance = context.HttpContext.Request.Path
            };
            pd.Extensions["traceId"] = context.HttpContext.TraceIdentifier;

            return new BadRequestObjectResult(pd)
            {
                ContentTypes = { "application/problem+json" }
            };
        };
    });

Отключение автоматической валидации ModelState

// Иногда нужно отключить автоматический 400 и проверять ModelState вручную
builder.Services.AddControllers()
    .ConfigureApiBehaviorOptions(options =>
    {
        options.SuppressModelStateInvalidFilter = true;
        // ↑ Отключает автоматический ModelStateInvalidFilter.
        //   Теперь [ApiController] НЕ будет возвращать 400 при невалидном ModelState.
        //   Нужно проверять ModelState.IsValid вручную в контроллере.
    });

// В контроллере:
[HttpPost("register")]
public async Task<IActionResult> Register([FromBody] RegisterDto dto)
{
    if (!ModelState.IsValid)
    {
        // Можно добавить дополнительные проверки перед возвратом ошибки
        var errors = ModelState.Where(e => e.Value?.Errors.Any() == true)
                               .ToDictionary(k => k.Key, v => v.Value!.Errors
                                   .Select(e => e.ErrorMessage).ToArray());

        return BadRequest(new { errors, traceId = HttpContext.TraceIdentifier });
    }

    var result = await _authService.RegisterAsync(dto);
    return Ok(result);
}

// Когда это нужно:
// - Нужна дополнительная логика перед возвратом ошибки (например, аудит-лог)
// - Ошибки валидации модели и бизнес-ошибки должны возвращаться в одном формате
// - Постепенная миграция Legacy-кода на [ApiController]
В WebMessenger регистрация валидации минимальна: RegisterDto и LoginDto используют [Required] + [MaxLength]/[MinLength], UpdateProfileDto — [EmailAddress], [Phone], [Url]. Межполевая валидация не применяется. Точки роста: добавить [Username] constraint для регистрации, IValidatableObject для составных запросов, и ConfigureApiBehaviorOptions для единообразного ProblemDetails с traceId.

4. Exception Handling

Этот раздел охватывает три взаимодополняющих инструмента: пользовательский middleware для централизованной обработки исключений, ручное создание расширенного ProblemDetails-ответа и управление детализацией ошибок через IHostEnvironment.

4.1 Custom Exception Handling Middleware

Что это. Custom Exception Middleware — middleware, который оборачивает весь дальнейший конвейер в try/catch и перехватывает необработанные исключения, превращая их в структурированный HTTP-ответ. Это централизованная альтернатива try/catch в каждом контроллере.

Проблема текущего подхода в WebMessenger. Каждый action в проекте содержит свой try/catch с StatusCode(500, new { message = "..." }). Это рабочий, но дублирующийся подход: 12 контроллеров × несколько action'ов = десятки однотипных блоков, которые возвращают ответы в разных форматах (не RFC 7807). Middleware решает это в одном месте.

Два пути в ASP.NET Core 8

Подход Интерфейс / Метод Когда использовать
Классический middleware Класс с InvokeAsync(HttpContext) Полный контроль над ответом, сложная логика
IExceptionHandler (ASP.NET Core 8) IExceptionHandler.TryHandleAsync() Цепочка обработчиков по типу исключения, чище DI

Вариант 1: Классический Exception Middleware

// Middleware: перехватывает все необработанные исключения и возвращает ProblemDetails
public class ExceptionHandlingMiddleware(
    RequestDelegate next,
    ILogger<ExceptionHandlingMiddleware> logger,
    IHostEnvironment env)
{
    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await next(context); // выполняем весь остальной конвейер
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Unhandled exception for {Method} {Path}",
                context.Request.Method, context.Request.Path);

            await HandleExceptionAsync(context, ex);
        }
    }

    private async Task HandleExceptionAsync(HttpContext context, Exception ex)
    {
        // Определяем статус-код и заголовок по типу исключения
        var (statusCode, title) = ex switch
        {
            NotFoundException          => (404, "Resource Not Found"),
            UnauthorizedAccessException => (401, "Unauthorized"),
            ForbiddenException         => (403, "Forbidden"),
            ValidationException        => (422, "Validation Failed"),   // бизнес-валидация
            ConflictException          => (409, "Conflict"),
            _                          => (500, "Internal Server Error")
        };

        var problem = new ProblemDetails
        {
            Status   = statusCode,
            Title    = title,
            Type     = $"https://tools.ietf.org/html/rfc9110#section-15.{statusCode / 100}.{statusCode % 100}",
            Instance = context.Request.Path,
            // Detail: в production — общее сообщение; в development — полное ex.Message
            Detail   = env.IsDevelopment() ? ex.Message : "An unexpected error occurred."
        };

        problem.Extensions["traceId"] = context.TraceIdentifier;

        // В development добавляем стек-трейс для отладки
        if (env.IsDevelopment())
            problem.Extensions["stackTrace"] = ex.StackTrace;

        context.Response.StatusCode  = statusCode;
        context.Response.ContentType = "application/problem+json";

        await context.Response.WriteAsJsonAsync(problem);
    }
}

// Регистрация в Program.cs — должен быть ПЕРВЫМ в конвейере
// (перехватывает исключения из всех последующих middleware)
app.UseMiddleware<ExceptionHandlingMiddleware>();

app.UseCors(FrontCors);
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();

Вариант 2: IExceptionHandler (ASP.NET Core 8)

// IExceptionHandler — новый интерфейс в .NET 8.
// Позволяет регистрировать несколько обработчиков по типу исключения.
// Если TryHandleAsync вернул false — передаёт управление следующему обработчику.

// Обработчик бизнес-исключений (известные ошибки)
public class DomainExceptionHandler(ILogger<DomainExceptionHandler> logger)
    : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        // Обрабатываем только известные доменные исключения
        if (exception is not (NotFoundException or ConflictException or ValidationException))
            return false; // false = передать следующему обработчику

        logger.LogWarning(exception, "Domain exception: {Message}", exception.Message);

        var (status, title) = exception switch
        {
            NotFoundException  => (404, "Not Found"),
            ConflictException  => (409, "Conflict"),
            ValidationException => (422, "Validation Error"),
            _                  => (400, "Bad Request")
        };

        var problem = new ProblemDetails
        {
            Status   = status,
            Title    = title,
            Detail   = exception.Message,
            Instance = httpContext.Request.Path
        };
        problem.Extensions["traceId"] = httpContext.TraceIdentifier;

        httpContext.Response.StatusCode = status;
        await httpContext.Response.WriteAsJsonAsync(problem, cancellationToken);

        return true; // true = обработано, дальше не передаём
    }
}

// Обработчик-fallback для всех остальных исключений
public class UnhandledExceptionHandler(
    ILogger<UnhandledExceptionHandler> logger,
    IHostEnvironment env)
    : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        logger.LogError(exception, "Unhandled exception");

        var problem = new ProblemDetails
        {
            Status   = 500,
            Title    = "Internal Server Error",
            Detail   = env.IsDevelopment() ? exception.Message : "An unexpected error occurred.",
            Instance = httpContext.Request.Path
        };
        problem.Extensions["traceId"] = httpContext.TraceIdentifier;

        if (env.IsDevelopment())
            problem.Extensions["stackTrace"] = exception.StackTrace;

        httpContext.Response.StatusCode = 500;
        await httpContext.Response.WriteAsJsonAsync(problem, cancellationToken);

        return true;
    }
}

// Регистрация нескольких IExceptionHandler в Program.cs
// Порядок регистрации = порядок вызова TryHandleAsync
builder.Services.AddExceptionHandler<DomainExceptionHandler>();    // 1. сначала доменные
builder.Services.AddExceptionHandler<UnhandledExceptionHandler>(); // 2. потом всё остальное

builder.Services.AddProblemDetails(); // нужен для UseExceptionHandler()

// В конвейере:
app.UseExceptionHandler(); // без аргументов — использует IExceptionHandler из DI
UseExceptionHandler() без аргументов (как в WebMessenger) работает только при наличии зарегистрированных IExceptionHandler-сервисов или AddProblemDetails(). Если ни того ни другого — выбросит InvalidOperationException при старте. WebMessenger использует AddProblemDetails(), поэтому UseExceptionHandler() корректно возвращает стандартный ProblemDetails для необработанных исключений.

Проблема текущего кода WebMessenger

4.2 ProblemDetails — расширенный ответ об ошибке вручную

Что это. Иногда нужно вернуть кастомный ProblemDetails не из middleware, а прямо из action-метода контроллера. Например, для бизнес-ошибок с дополнительными полями, которые важны для клиента (код ошибки, ссылка на документацию, список конфликтующих объектов).

Ручное создание ProblemDetails в контроллере

// Вариант 1: Через Problem() — метод ControllerBase
[HttpPost("login")]
public async Task<IActionResult> Login([FromBody] LoginDto dto)
{
    var user = await _userService.FindUserByUsernameAsync(dto.Username);
    if (user is null || !_authService.ValidateUserCredentials(user, dto.Password))
    {
        // Problem() создаёт ObjectResult с ProblemDetails и Content-Type: application/problem+json
        return Problem(
            statusCode: StatusCodes.Status401Unauthorized,
            title:      "Authentication Failed",
            detail:     "Invalid username or password.",
            instance:   HttpContext.Request.Path,
            type:       "https://tools.ietf.org/html/rfc9110#section-15.5.2"
        );
    }

    var token = _authService.GenerateJwtToken(user);
    // ...
    return Ok(new { token });
}

// Вариант 2: Ручная сборка ProblemDetails с кастомными extension-полями
[HttpPost("direct/{userId:guid}/messages")]
public async Task<IActionResult> SendMessage(Guid userId, [FromBody] SendMessageRequest request)
{

        problem.Extensions["addContactLink"]  = Url.Action("AddContact", "Contacts",

        {

    var message = await _chatService.SendMessageAsync(_currentUser.Id, userId, request);

}

Что это. IHostEnvironment — сервис, предоставляющий информацию о текущей среде запуска: имя среды (Development, Staging, Production), путь к контенту, имя приложения. Центральный инструмент для условной логики «в dev показываем больше, в prod — меньше».

Условный ProblemDetails в зависимости от среды

// Паттерн: детализация ошибки зависит от среды
public class GlobalExceptionHandler(
    ILogger<GlobalExceptionHandler> logger,
    IHostEnvironment env) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken ct)
    {
        logger.LogError(exception, "Unhandled exception at {Path}", httpContext.Request.Path);

        var problem = new ProblemDetails
        {
            Status   = 500,
            Title    = "Internal Server Error",
            Instance = httpContext.Request.Path,
            Type     = "https://tools.ietf.org/html/rfc9110#section-15.6.1"
        };

        if (env.IsDevelopment())
        {
            // Development: полная информация для разработчика
            problem.Detail = exception.Message;
            problem.Extensions["exceptionType"] = exception.GetType().FullName;
            problem.Extensions["stackTrace"]    = exception.StackTrace?.Split('\n')
                                                            .Take(15)          // первые 15 строк
                                                            .ToArray();
            problem.Extensions["innerException"] = exception.InnerException?.Message;
        }
        else
        {
            // Production: никаких деталей реализации клиенту
            problem.Detail = "An unexpected error occurred. Please try again later.";
            // traceId позволяет найти запись в логах, не раскрывая детали
        }

        problem.Extensions["traceId"] = httpContext.TraceIdentifier;

        httpContext.Response.StatusCode  = 500;
        httpContext.Response.ContentType = "application/problem+json";
        await httpContext.Response.WriteAsJsonAsync(problem, ct);

        return true;
    }
}

5.1 API Versioning — стратегии версионирования

Что это. API Versioning — механизм поддержки нескольких версий API одновременно. При изменении контрактов DTO, маршрутов или поведения существующие клиенты продолжают работать со старой версией, новые — переходят на новую.

Пакет. Asp.Versioning.Http (официальный, от Microsoft) — заменил устаревший Microsoft.AspNetCore.Mvc.Versioning.

Три стратегии: сравнение

Стратегия Пример запроса Плюсы Минусы
URL Path GET /api/v1/users Явно видна в URL, легко кешировать, удобна для документации Нарушает REST (URI должен идентифицировать ресурс, не версию)
HTTP Header X-Api-Version: 2.0 Чистые URI, клиент явно указывает версию Сложнее тестировать в браузере, не виден в кешах
Query String GET /api/users?api-version=2.0 Легко тестировать, поддерживается без заголовков Засоряет URI, нестандартно для REST

Настройка — все три стратегии одновременно

// Установка пакета:
// dotnet add package Asp.Versioning.Mvc
// dotnet add package Asp.Versioning.Mvc.ApiExplorer (для Swagger)

// Program.cs
builder.Services
    .AddApiVersioning(options =>
    {
        // Версия по умолчанию — используется если клиент не указал версию
        options.DefaultApiVersion = new ApiVersion(1, 0);
        options.AssumeDefaultVersionWhenUnspecified = true;
        // ↑ Если версия не указана — используется DefaultApiVersion (v1.0)
        //   Без этого — 400 "API version was not specified"

        // Возвращать поддерживаемые версии в заголовках ответа:
        // api-supported-versions: 1.0, 2.0
        // api-deprecated-versions: 1.0 (если версия помечена Deprecated)
        options.ReportApiVersions = true;

        // Читать версию из трёх источников одновременно (приоритет по порядку):
        options.ApiVersionReader = ApiVersionReader.Combine(
            new UrlSegmentApiVersionReader(),          // /api/v{version}/
            new HeaderApiVersionReader("X-Api-Version"), // X-Api-Version: 2.0
            new QueryStringApiVersionReader("api-version") // ?api-version=2.0
        );
    })
    .AddApiExplorer(options =>
    {
        // Формат группировки версий в Swagger UI:
        // "v1", "v2" — используется как groupName в IApiVersionDescriptionProvider
        options.GroupNameFormat = "'v'VVV";
        options.SubstituteApiVersionInUrl = true;
        // ↑ Заменяет {version} в шаблоне маршрута на реальный номер версии в Swagger
    });

[Route("api/v{version:apiVersion}/users")]
public class UsersV1Controller : ControllerBase
{

    public IActionResult GetAll() => Ok(new { version = "v1", users = new[] { "alice" } });

}

[ApiController]

[Route("api/v{version:apiVersion}/users")]
public class UsersV2Controller : ControllerBase
{

    // Новая версия возвращает PagedResult вместо массива

// GET /api/v1/users → UsersV1Controller.GetAll()
// GET /api/v2/users → UsersV2Controller.GetAll()

Версионирование в одном контроллере через MapToApiVersion

// Один контроллер поддерживает несколько версий — удобно для небольших изменений
[ApiController]
[ApiVersion("1.0")]
[ApiVersion("2.0")]
[ApiVersion("1.0", Deprecated = true)]  // 1.0 — устаревшая (ещё работает, но предупреждение)

    [MapToApiVersion("2.0")]
    public IActionResult GetChatsV2([FromQuery] int limit = 20, [FromQuery] DateTime? before = null)
        => Ok(new { version = "v2", limit, before, chats = Array.Empty<object>() });

    // Общий endpoint для всех версий — без [MapToApiVersion]
    [HttpGet("{chatId:guid}/health")]
    public IActionResult HealthCheck(Guid chatId) => Ok();

5.2 Content Negotiation

Что это. Content Negotiation — механизм, при котором клиент через заголовок Accept сообщает, в каком формате хочет получить ответ (application/json, application/xml, text/csv), а сервер выбирает подходящий форматтер.

По умолчанию ASP.NET Core поддерживает только JSON (System.Text.Json). XML, CSV и другие форматы требуют дополнительной настройки.

// Добавление XML-форматтера:
builder.Services.AddControllers()
    .AddXmlSerializerFormatters()      // XmlSerializer (атрибуты из System.Xml.Serialization)
    // или
    .AddXmlDataContractSerializerFormatters(); // DataContractSerializer (атрибуты [DataContract])

// После этого:
// Accept: application/json → JSON-ответ (по умолчанию)
// Accept: application/xml  → XML-ответ
// Accept: text/html        → 406 Not Acceptable (нет форматтера)

// Ограничение форматов через [Produces] и [Consumes]:
[ApiController]
[Route("api/reports")]
[Produces("application/json", "application/xml")]
// ↑ Контроллер поддерживает только JSON и XML — другие Accept → 406
public class ReportsController : ControllerBase
{
    [HttpPost]
    [Consumes("application/json")]
    // ↑ Принимает только JSON в теле запроса — другой Content-Type → 415 Unsupported Media Type
    public IActionResult CreateReport([FromBody] ReportRequest request) => Ok();

    // Форматтер можно переопределить на уровне action:
    [HttpGet("{id}")]
    [Produces("text/csv")]
    public IActionResult ExportCsv(int id)
    {
        var csv = "id,name,value\n1,test,42";
        return Content(csv, "text/csv");
        // Content() возвращает ContentResult, Content-Type берётся из аргумента
    }
}

// Настройка Content Negotiation глобально:
builder.Services.AddControllers(options =>
{
    // RespectBrowserAcceptHeader: браузеры посылают широкий Accept (text/html, */*)
    // По умолчанию ASP.NET Core игнорирует */*, всегда возвращает JSON.
    // Если true — попытается подобрать форматтер под браузерный Accept.
    options.RespectBrowserAcceptHeader = true;

    // ReturnHttpNotAcceptable: если нет форматтера для Accept → 406
    // По умолчанию false: возвращает JSON даже если клиент просит XML.
    options.ReturnHttpNotAcceptable = true;
});

5.3 Response Compression и Request Decompression

Что это. Response Compression — сжатие тела HTTP-ответа на стороне сервера (Gzip / Brotli). Уменьшает объём передаваемых данных, ускоряет загрузку для клиентов с медленным соединением. Request Decompression — обратный процесс: сервер принимает сжатые тела запросов (Content-Encoding: gzip).

Когда не нужно. Если перед API стоит nginx или другой реверс-прокси — сжатие лучше делать на уровне прокси. В Docker-окружении WebMessenger nginx (если используется) может взять эту задачу на себя. Встроенное сжатие в Kestrel имеет смысл при прямом выходе в интернет.

Response Compression

// Program.cs — настройка сжатия ответов
builder.Services.AddResponseCompression(options =>
{
    // По умолчанию сжатие применяется только к HTTP (не HTTPS).
    // В HTTPS-окружении сжатие создаёт уязвимость BREACH/CRIME для чувствительных данных.
    // Если API возвращает только JSON без секретов — можно включить для HTTPS:
    options.EnableForHttps = true;

    options.Providers.Add<BrotliCompressionProvider>();   // лучший уровень сжатия, современные браузеры
    options.Providers.Add<GzipCompressionProvider>();     // широкая совместимость

    // MIME-типы для сжатия (дополнительно к встроенным: text/plain, text/html, application/json...)
    options.MimeTypes = ResponseCompressionDefaults.MimeTypes.Concat(
        ["application/problem+json", "application/xml"]);
});

// Настройка уровня сжатия:
builder.Services.Configure<BrotliCompressionProviderOptions>(options =>
{
    options.Level = CompressionLevel.Optimal;
    // Optimal — лучшее соотношение размер/скорость (рекомендуется для API)
    // Fastest — быстрее, хуже сжатие
    // SmallestSize — максимальное сжатие, медленнее
});

builder.Services.Configure<GzipCompressionProviderOptions>(options =>
{
    options.Level = CompressionLevel.Optimal;
});

// Применение middleware — до MapControllers, после UseRouting:
app.UseResponseCompression();
app.MapControllers();

// Результат: клиент с Accept-Encoding: br, gzip получит сжатый ответ.
// Клиент без Accept-Encoding — обычный ответ.

Request Decompression (ASP.NET Core 7+)

// Позволяет клиентам отправлять сжатые запросы с Content-Encoding: gzip/br/deflate.
// Полезно для мобильных клиентов, загружающих большие payload.

builder.Services.AddRequestDecompression();

app.UseRequestDecompression(); // до UseRouting

// Клиент:
// POST /api/chats/messages
// Content-Encoding: gzip
// [gzip-сжатое тело запроса]
//
// Middleware автоматически разжимает тело перед передачей в контроллер.
// Контроллер получает обычный JSON как будто сжатия не было.

5.4 Rate Limiting и Throttling

Что это. Rate Limiting — ограничение количества запросов за единицу времени. Защищает API от перегрузки, brute-force атак на аутентификацию, злоупотребления публичными эндпоинтами. В ASP.NET Core 7+ встроена поддержка через Microsoft.AspNetCore.RateLimiting.

Четыре алгоритма

Алгоритм Описание Когда использовать
Fixed Window N запросов за фиксированное окно (1 мин, 1 час). Сброс по таймеру. Простые ограничения, публичный API
Sliding Window N запросов за скользящее окно. Более равномерная нагрузка. Когда Fixed Window даёт burst при сбросе
Token Bucket Ведро с токенами, пополняется равномерно. Допускает короткие burst'ы. API с допустимыми всплесками (загрузка файлов)
Concurrency Ограничение числа одновременных запросов (не per-time). Защита от перегрузки при долгих операциях

Конфигурация Rate Limiting

// Program.cs
builder.Services.AddRateLimiter(options =>
{
    // Глобальный обработчик при превышении лимита:
    options.OnRejected = async (context, ct) =>
    {
        context.HttpContext.Response.StatusCode = 429; // Too Many Requests

        if (context.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfter))
        {
            // Заголовок Retry-After: сколько секунд ждать перед повтором
            context.HttpContext.Response.Headers.RetryAfter =
                ((int)retryAfter.TotalSeconds).ToString();
        }

        await context.HttpContext.Response.WriteAsJsonAsync(new ProblemDetails
        {
            Status = 429,
            Title  = "Too Many Requests",
            Detail = "Request rate limit exceeded. Please slow down."
        }, ct);
    };

    // Политика 1: Fixed Window — для анонимных запросов
    options.AddFixedWindowLimiter("anonymous", limiterOptions =>
    {
        limiterOptions.PermitLimit         = 20;         // 20 запросов
        limiterOptions.Window              = TimeSpan.FromMinutes(1); // за 1 минуту
        limiterOptions.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        limiterOptions.QueueLimit           = 5;         // буфер ожидания 5 запросов
    });

    // Политика 2: Sliding Window — для аутентификации (жёстче)
    options.AddSlidingWindowLimiter("auth", limiterOptions =>
    {
        limiterOptions.PermitLimit         = 5;           // 5 попыток
        limiterOptions.Window              = TimeSpan.FromMinutes(5); // за 5 минут
        limiterOptions.SegmentsPerWindow   = 5;           // окно делится на 5 сегментов
        limiterOptions.QueueLimit          = 0;           // без очереди — сразу 429
    });

    // Политика 3: Token Bucket — для general API endpoints
    options.AddTokenBucketLimiter("api", limiterOptions =>
    {
        limiterOptions.TokenLimit          = 100;         // максимум токенов в ведре
        limiterOptions.ReplenishmentPeriod = TimeSpan.FromSeconds(10); // каждые 10 сек
        limiterOptions.TokensPerPeriod     = 20;          // добавляем 20 токенов
        limiterOptions.AutoReplenishment   = true;
    });

    // Политика 4: Concurrency — ограничение одновременных запросов
    options.AddConcurrencyLimiter("upload", limiterOptions =>
    {
        limiterOptions.PermitLimit = 3;   // не более 3 одновременных загрузок файлов
        limiterOptions.QueueLimit  = 10;
    });
});

// Применение middleware:
app.UseRateLimiter(); // после UseRouting, до MapControllers
app.MapControllers();

Per-User Rate Limiting (партиционирование)

// Стандартные политики применяются globally.
// Для ограничения "5 запросов на пользователя" нужно партиционирование:
options.AddPolicy("per-user", httpContext =>
{
    // Ключ партиции: userId для аутентифицированных, IP для анонимных
    var userId = httpContext.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
    var partitionKey = userId ?? httpContext.Connection.RemoteIpAddress?.ToString() ?? "unknown";

    return RateLimitPartition.GetFixedWindowLimiter(
        partitionKey: partitionKey,
        factory: _ => new FixedWindowRateLimiterOptions
        {
            PermitLimit = 100,
            Window      = TimeSpan.FromMinutes(1)
        });
});

5.5 OpenAPI/Swagger: ProducesResponseType и Security Definitions

Что это. Swagger/OpenAPI — автоматическая генерация интерактивной документации API. [ProducesResponseType] описывает возможные коды ответа и типы для каждого action. Security Definitions позволяют использовать JWT-аутентификацию прямо из Swagger UI.

ProducesResponseType — документирование ответов

// Без атрибутов Swagger показывает только "200 OK" без описания схемы.
// ProducesResponseType добавляет точную документацию.

[HttpGet("{chatId:guid}/messages")]
[ProducesResponseType(typeof(PagedResult<ChatMessageDto>), StatusCodes.Status200OK)]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status401Unauthorized)]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status403Forbidden)]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status404NotFound)]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status500InternalServerError)]
public async Task<ActionResult<PagedResult<ChatMessageDto>>> GetMessages(Guid chatId) { ... }

// С именованными статус-кодами вместо чисел:
[ProducesResponseType<PagedResult<ChatMessageDto>>(StatusCodes.Status200OK)]
// ↑ Generic-синтаксис (C# 11) — эквивалент typeof(), но короче

// На уровне контроллера — применяется ко всем action'ам:
[ApiController]

// OperationFilter — помечает анонимные endpoints как открытые
public class AuthorizeCheckOperationFilter : IOperationFilter
{
    public void Apply(OpenApiOperation operation, OperationFilterContext context)
    {
        // Проверяем, есть ли [AllowAnonymous] на action или контроллере
        var hasAllowAnonymous = context.MethodInfo
            .GetCustomAttributes(true)
            .OfType<AllowAnonymousAttribute>()
            .Any()
            || context.MethodInfo.DeclaringType!
            .GetCustomAttributes(true)
            .OfType<AllowAnonymousAttribute>()
            .Any();

        if (hasAllowAnonymous)
        {
            // Убираем требование безопасности для анонимных endpoints
            operation.Security.Clear();
            return;
        }

        // Для защищённых endpoints добавляем иконку замка в Swagger UI
        operation.Security = [
            new OpenApiSecurityRequirement
            {
                {
                    new OpenApiSecurityScheme
                    {
                        Reference = new OpenApiReference
                        {
                            Type = ReferenceType.SecurityScheme,
                            Id   = "Bearer"
                        }
                    },
                    Array.Empty<string>()
                }
            }
        ];
    }
}

6. Caching

ASP.NET Core предоставляет три уровня кэширования: Response Caching (HTTP-кэш на уровне конвейера), Distributed Cache (разделяемое хранилище между экземплярами) и Output Caching (новый встроенный кэш в .NET 7+). Они решают разные задачи и часто используются совместно.

6.1 Response Caching

Что это. Response Caching работает на уровне HTTP: добавляет к ответу заголовки Cache-Control, Vary, Expires, которые инструктируют клиента и промежуточные прокси (CDN, nginx) о том, можно ли кэшировать ответ и на сколько. Middleware UseResponseCaching дополнительно кэширует ответы на стороне сервера.

Ключевое ограничение. Response Caching работает только для ответов на GET и HEAD запросы, без тела ответа или Set-Cookie заголовка, и только если запрос не содержит Authorization. Поскольку WebMessenger использует JWT Bearer в заголовке Authorization — все [Authorize] endpoints не будут кэшироваться на стороне сервера через UseResponseCaching.

Настройка и применение

// Program.cs
builder.Services.AddResponseCaching(options =>
{
    options.MaximumBodySize    = 64 * 1024; // максимальный размер кэшируемого ответа (64 KB)
    options.UseCaseSensitivePaths = false;  // /api/Users и /api/users — один кэш
    options.SizeLimit = 100 * 1024 * 1024; // общий лимит кэша в памяти (100 MB)
});

// В конвейере (ПОРЯДОК ВАЖЕН):
app.UseResponseCaching();   // до UseAuthorization, до MapControllers
app.UseAuthorization();
app.MapControllers();

// Кэшировать на 60 секунд (публичный — можно кэшировать в CDN):

[ResponseCache(Duration = 60, Location = ResponseCacheLocation.Any, VaryByQueryKeys = new[] { "*" })]

{

    var profile = await _userService.GetUserProfileAsync(id);

}

// Кэшировать только в браузере клиента, не в прокси:

{

    // Каждый пользователь получает свой кэш
    return Ok(await _userService.GetUserProfileAsync(_currentUser.Id));
}

// Запретить кэширование полностью:

[ResponseCache(NoStore = true)]

6.2 Distributed Cache — IDistributedCache

Что это. IDistributedCache — интерфейс для разделяемого кэша, доступного всем экземплярам приложения. В отличие от IMemoryCache (только в памяти одного процесса), Distributed Cache работает с Redis, SQL Server или другим внешним хранилищем.

Когда использовать Distributed вместо Memory.

IMemoryCache IDistributedCache
Хранилище RAM одного процесса Redis, SQL Server, внешнее
Масштабирование Не работает при нескольких экземплярах Разделяется между всеми экземплярами
Выживаемость Теряется при рестарте Сохраняется между рестартами (зависит от хранилища)
Сериализация Объекты напрямую Только byte[] / string (нужна сериализация)
Latency Наносекунды Миллисекунды (сеть к Redis)

Работа с IDistributedCache

// Пример: кэширование профиля пользователя (абстрактный сервис для WebMessenger)
public class CachedUserService(
    IUserService inner,
    IDistributedCache cache,
    ILogger<CachedUserService> logger)
    : IUserService
{
    private static readonly DistributedCacheEntryOptions _profileCacheOptions =
        new DistributedCacheEntryOptions()
            .SetAbsoluteExpiration(TimeSpan.FromMinutes(5))  // удалить через 5 мин в любом случае
            .SetSlidingExpiration(TimeSpan.FromMinutes(2));  // обновить TTL при обращении
            // ↑ Sliding + Absolute: обновляем TTL при каждом чтении, но не дольше 5 мин

    public async Task<UserProfileDto?> GetUserProfileAsync(Guid userId)
    {
        var cacheKey = $"user:profile:{userId}";

        // 1. Пробуем получить из кэша
        var cached = await cache.GetStringAsync(cacheKey);
        if (cached is not null)
        {
            logger.LogDebug("Cache HIT for user profile {UserId}", userId);
            return JsonSerializer.Deserialize<UserProfileDto>(cached);
        }

        // 2. Кэш MISS → идём в БД
        logger.LogDebug("Cache MISS for user profile {UserId}", userId);
        var profile = await inner.GetUserProfileAsync(userId);

        if (profile is not null)
        {
            // 3. Сохраняем в кэш
            await cache.SetStringAsync(
                cacheKey,
                JsonSerializer.Serialize(profile),
                _profileCacheOptions);
        }

        return profile;
    }

    public async Task<UserProfileDto?> UpdateUserProfileAsync(Guid userId, UpdateProfileDto dto)
    {
        var result = await inner.UpdateUserProfileAsync(userId, dto);

        // 4. Инвалидация кэша при обновлении
        await cache.RemoveAsync($"user:profile:{userId}");
        logger.LogDebug("Cache invalidated for user profile {UserId}", userId);

        return result;
    }

    // Остальные методы делегируются inner без кэширования...
}

// Регистрация Decorator-паттерна:
builder.Services.AddScoped<UserService>();    // конкретная реализация
builder.Services.AddScoped<IUserService>(sp => // декоратор с кэшем
    new CachedUserService(
        sp.GetRequiredService<UserService>(),
        sp.GetRequiredService<IDistributedCache>(),
        sp.GetRequiredService<ILogger<CachedUserService>>()));

6.3 Output Caching

Что это. Output Caching — новый механизм в ASP.NET Core 7+, встроенный непосредственно в конвейер. В отличие от Response Caching, он кэширует ответ на стороне сервера независимо от заголовков Authorization, Cache-Control и Set-Cookie. Это делает его применимым для аутентифицированных endpoints.

Ключевые отличия от Response Caching:

Response Caching Output Caching
Уровень управления HTTP-заголовки (Cache-Control) Серверный, независимо от заголовков
Работает с Authorization Нет Да (если явно настроено)
Инвалидация Только по TTL По TTL + программная инвалидация по тегам
Хранилище Только память Память или кастомный ICacheStore
Доступен с версии ASP.NET Core 1.0 ASP.NET Core 7.0

Настройка Output Caching

// Program.cs
builder.Services.AddOutputCache(options =>
{
    // Базовая политика — применяется ко всем endpoints без явной политики
    options.AddBasePolicy(policy =>
        policy.Expire(TimeSpan.FromSeconds(10)));

    // Именованная политика для профилей пользователей
    options.AddPolicy("UserProfiles", policy =>
        policy
            .Expire(TimeSpan.FromMinutes(5))
            .SetVaryByQuery("userId")      // разные кэши для разных ?userId=...
            .Tag("users"));                // тег для групповой инвалидации

    // Политика с варьированием по заголовку
    options.AddPolicy("LocalizedContent", policy =>
        policy
            .Expire(TimeSpan.FromHours(1))
            .SetVaryByHeader("Accept-Language")
            .Tag("localized", "content"));

    // Политика без кэширования (например, для real-time данных)
    options.AddPolicy("NoCache", policy =>
        policy.NoCache());
});

// Middleware:
app.UseOutputCache(); // после UseRouting, до UseAuthorization и MapControllers

public class UserService(

Практическое применение в WebMessenger: Хорошие кандидаты для Output Cache — GET /api/users/profile/{id} (профиль другого пользователя, редко меняется), GET /api/users?query=... 7. Security

Этот раздел охватывает два продвинутых аспекта безопасности: механизм защиты от CSRF-атак через Anti-Forgery токены и тонкую настройку Content Security Policy как инструмента защиты от XSS.

7.1 CSRF и XSS — природа атак

Прежде чем разбирать защиты, нужно точно понять, от чего они защищают.

Атака Механизм Что крадёт / делает Основная защита
XSS (Cross-Site Scripting) Инъекция скрипта в страницу — браузер выполняет его в контексте вашего сайта Читает DOM, cookies (если не HttpOnly), localStorage, отправляет данные на сервер атакующего Content Security Policy, HttpOnly cookies, экранирование вывода
CSRF (Cross-Site Request Forgery) Вредоносный сайт заставляет браузер отправить запрос к вашему API от имени уже аутентифицированного пользователя Выполняет действия от имени пользователя (перевод денег, смена email, удаление данных) Anti-Forgery токены, SameSite cookies, проверка Origin/Referer

Ключевая связь. CSRF работает потому, что браузер автоматически прикрепляет cookies к запросам. Если токен аутентификации хранится в cookie (как auth-token в WebMessenger), CSRF потенциально применим. Если токен хранится только в localStorage и передаётся через Authorization: Bearer, CSRF неприменим — JavaScript вредоносного сайта не может прочитать localStorage другого домена.

7.2 ValidateAntiForgeryToken — защита от CSRF

Что это. Anti-Forgery механизм генерирует уникальный токен для каждой сессии пользователя и требует его присутствия при изменяющих запросах (POST, PUT, DELETE). Вредоносный сайт не может получить этот токен (SOP — Same Origin Policy), поэтому не может сформировать валидный запрос.

Как работает двойная проверка.

  1. Сервер генерирует anti-forgery токен и отдаёт его двумя способами: в cookie .AspNetCore.Antiforgery.* и в hidden-поле формы (или HTTP-заголовке X-CSRF-TOKEN для SPA)
  2. При запросе клиент присылает оба значения: cookie (браузер делает это автоматически) и токен из формы/заголовка
  3. Сервер сверяет оба значения. Если они совпадают — запрос легитимный (только ваш JavaScript мог прочитать токен из формы/заголовка)

Регистрация и базовое использование

// Program.cs
builder.Services.AddAntiforgery(options =>
{
    // Имя cookie с server-side токеном:
    options.Cookie.Name     = "XSRF-TOKEN";

    options.Cookie.HttpOnly = false;

    options.Cookie.SameSite     = SameSiteMode.Strict;

    // Angular автоматически читает XSRF-TOKEN cookie и отправляет X-XSRF-TOKEN заголовок.
    // React/Next.js — нужно реализовать вручную.

// Для SPA (Next.js, React) антифоржери токен нужно явно отдать клиенту.
// Типичный паттерн — специальный endpoint:

[ApiController]
[Route("api/antiforgery")]
public class AntiforgeryController(IAntiforgery antiforgery) : ControllerBase
{
    [HttpGet("token")]
    [IgnoreAntiforgeryToken] // GET-запрос не нуждается в проверке
    public IActionResult GetToken()
    {
        // Генерируем токен и устанавливаем cookie + возвращаем значение в теле
        var tokens = antiforgery.GetAndStoreTokens(HttpContext);

CSRF и WebMessenger: применимость

// WebMessenger использует два способа передачи JWT:
// 1. Authorization: Bearer {token} — в заголовке запроса (JWT из localStorage/памяти)
// 2. Cookie "auth-token" (HttpOnly: true, SameSite: Lax) — установлена при логине

// ─── Анализ CSRF-риска ───────────────────────────────────────────────

// REST API (контроллеры) — запросы от Next.js клиента:
// Next.js отправляет запросы через fetch() с явным Authorization: Bearer заголовком.
// Вредоносный сайт НЕ МОЖЕТ установить Authorization заголовок через форму/img/fetch без CORS.
// → CSRF через REST API неприменим при использовании Bearer заголовка.

// Уязвимость остаётся только если:
// - Клиент использует ИСКЛЮЧИТЕЛЬНО cookie "auth-token" без Authorization заголовка
// - Запрос отправляется как простая форма (без кастомных заголовков)

// SignalR хаб — WebSocket соединение:
// WebSocket не подвержен классическому CSRF (браузер не добавляет cookies автоматически
// при установке WebSocket соединения на другой домен).
// WebMessenger использует JWT из query-string или cookie для аутентификации хаба.

// ─── Итог ───────────────────────────────────────────────────────────
// SameSite=Lax на "auth-token" cookie уже даёт базовую защиту:
// форма с другого сайта не отправит cookie (cross-site non-navigational request).
// Для полной защиты — дополнительно применить [AutoValidateAntiforgeryToken]
// если когда-либо потребуется поддержать HTML-форм клиенты.
JWT + Bearer vs Anti-Forgery. Если API используется исключительно через JavaScript-клиент с Authorization: Bearer заголовком — Anti-Forgery токены не нужны. CSRF работает через автоматическую отправку cookies браузером; заголовки браузер произвольно не добавляет. Именно поэтому SPA-архитектуры с JWT в заголовке изначально защищены от CSRF без дополнительных мер.

7.3 Content Security Policy — продвинутая настройка

Что это. CSP — заголовок Content-Security-Policy, который инструктирует браузер о том, из каких источников разрешена загрузка ресурсов (скриптов, стилей, изображений, шрифтов, WebSocket-соединений и т.д.). Основной инструмент защиты от XSS на стороне браузера.

Принцип работы. Даже если атакующий внедрил <script src="https://evil.com/steal.js"> или inline-скрипт — браузер заблокирует выполнение, если оно не разрешено CSP политикой.

script-src JavaScript (<script>, eval, inline) 'self' (без 'unsafe-inline'!)
style-src CSS (<style>, style=, <link rel=stylesheet>) 'self' 'unsafe-inline' (Tailwind требует inline)
img-src Изображения 'self' data: https://res.cloudinary.com
font-src Шрифты (@font-face) 'self' https://fonts.gstatic.com

base-uri <base> тег 'self' (защита от base-URI hijacking)

Nonce-based CSP — безопасные inline-скрипты

// Проблема 'unsafe-inline': разрешает ВСЕ inline-скрипты, включая инъецированные XSS.
// Решение — nonce: уникальный токен на каждый запрос, который разрешает
// только конкретные inline-скрипты, содержащие этот nonce.

// ASP.NET Core: генерация nonce в middleware
public class CspNonceMiddleware(RequestDelegate next)
{
    public async Task InvokeAsync(HttpContext context)
    {
        // Генерируем случайный nonce для каждого запроса
        var nonce = Convert.ToBase64String(RandomNumberGenerator.GetBytes(16));

        // Сохраняем в Items для использования в представлениях/компонентах
        context.Items["CspNonce"] = nonce;

        context.Response.Headers.Append(
            "Content-Security-Policy",
            $"default-src 'self'; " +
            $"script-src 'self' 'nonce-{nonce}'; " + // разрешить только скрипты с этим nonce
            $"style-src 'self' 'nonce-{nonce}'; " +
            $"object-src 'none'; " +
            $"base-uri 'self';"
        );

        await next(context);
    }
}

8. Globalization and Localization

Globalization — подготовка приложения к работе с разными культурами (форматы дат, чисел, валют). Localization — перевод текстовых ресурсов (сообщений, ошибок, UI-текстов) на разные языки. В ASP.NET Core это два связанных, но независимых механизма.

8.1 Настройка сервисов — AddLocalization и AddRequestLocalization

Что это. AddLocalization() регистрирует сервисы для работы со строковыми ресурсами (IStringLocalizer, IStringLocalizerFactory). AddRequestLocalization() / AddRequestLocalizationOptions() настраивает стратегию определения культуры из входящего запроса.

Полная регистрация

// Program.cs
builder.Services.AddLocalization(options =>
{
    // ResourcesPath: папка где лежат .resx файлы (относительно корня проекта)
    // По умолчанию: корень проекта (рядом с .cs файлами)
    // Рекомендуется: отдельная папка "Resources"
    options.ResourcesPath = "Resources";
});

// AddRequestLocalizationOptions — конфигурация определения культуры из запроса
builder.Services.AddRequestLocalization(options =>
{
    // Поддерживаемые культуры — только они будут приняты:
    var supportedCultures = new[]
    {
        new CultureInfo("en"),    // English (нейтральная — без региона)
        new CultureInfo("en-US"), // English (United States)
        new CultureInfo("ru"),    // Russian
        new CultureInfo("ru-RU"), // Russian (Russia)
        new CultureInfo("de"),    // German
    };

    // DefaultRequestCulture — культура при отсутствии информации в запросе
    options.DefaultRequestCulture = new RequestCulture("en");

    // SupportedCultures: для Globalization — форматы дат, чисел, валют
    options.SupportedCultures = supportedCultures;

    // SupportedUICultures: для Localization — поиск строковых ресурсов (.resx)
    // Обычно совпадает с SupportedCultures
    options.SupportedUICultures = supportedCultures;

    // Стратегии определения культуры (порядок = приоритет):
    options.RequestCultureProviders = new List<IRequestCultureProvider>
    {
        new QueryStringRequestCultureProvider(),
        // ?culture=ru&ui-culture=ru → принять из query string
        // Удобно для тестирования: /api/greet?culture=ru

        new CookieRequestCultureProvider(),
        // Cookie ".AspNetCore.Culture" → принять сохранённую культуру
        // Удобно для web-приложений с настройкой языка

        new AcceptLanguageHeaderRequestCultureProvider(),
        // Accept-Language: ru-RU,ru;q=0.9,en;q=0.8 → принять из заголовка браузера
        // Самый "HTTP-стандартный" источник для API
    };
});

8.2 Fallback Culture

Что это. Fallback Culture определяет, что делать когда запрошенная культура (Accept-Language: fr) не поддерживается приложением. Вместо ошибки возвращается «наилучший доступный» вариант.

Цепочка fallback. Если запрошена культура ru-RU, а она не поддерживается, ASP.NET Core пробует:

  1. Точное совпадение: ru-RU
  2. Нейтральная культура: ru
  3. Культура по умолчанию: en (DefaultRequestCulture)

Настройка Fallback

builder.Services.AddRequestLocalization(options =>
{
    options.DefaultRequestCulture = new RequestCulture("en");
    // ↑ Финальный fallback: если ничего не подошло — используем "en"

    options.SupportedCultures   = [new CultureInfo("en"), new CultureInfo("ru")];
    options.SupportedUICultures = [new CultureInfo("en"), new CultureInfo("ru")];

    // FallbackToParentCultures: если "ru-RU" не поддерживается,
    // попробовать родительскую "ru" (true по умолчанию)
    options.FallbackToParentCultures   = true;
    options.FallbackToParentUICultures = true;

    // ApplyCurrentCultureToResponseHeaders: добавлять Content-Language заголовок к ответу
    // Content-Language: ru — браузер/клиент знает на каком языке ответ
    options.ApplyCurrentCultureToResponseHeaders = true;
});

// Пример цепочки fallback при FallbackToParentCultures = true:
// Accept-Language: de → "de" не поддерживается → "de" нет родителя в supported → DefaultRequestCulture "en"
// Accept-Language: ru-RU → "ru-RU" не поддерживается → пробуем родителя "ru" → "ru" поддерживается ✓
// Accept-Language: fr → "fr" не поддерживается → нет родителя в supported → DefaultRequestCulture "en"

Кастомный IRequestCultureProvider

// Определение культуры из JWT Claims — например, пользователь выбрал язык при регистрации
public class ClaimsRequestCultureProvider : IRequestCultureProvider
{
    public Task<ProviderCultureResult?> DetermineProviderCultureResult(HttpContext httpContext)
    {
        // Пытаемся получить preferred language из claims пользователя
        var languageClaim = httpContext.User.FindFirst("preferred_language");

        if (languageClaim is null)
            return Task.FromResult<ProviderCultureResult?>(null);
            // null → передать следующему провайдеру в цепочке

        // Возвращаем культуру из claim
        return Task.FromResult<ProviderCultureResult?>(
            new ProviderCultureResult(languageClaim.Value));
    }
}

// Регистрация — ПЕРВЫМ в списке, чтобы user preference имел приоритет:
options.RequestCultureProviders = new List<IRequestCultureProvider>
{
    new ClaimsRequestCultureProvider(),       // 1. из JWT claim
    new QueryStringRequestCultureProvider(),  // 2. из ?culture=
    new AcceptLanguageHeaderRequestCultureProvider(), // 3. из Accept-Language
};

8.3 RequestLocalizationMiddleware — как определяется культура

Что происходит при каждом запросе. UseRequestLocalization() устанавливает middleware, который обходит список RequestCultureProviders по порядку. Первый провайдер, вернувший не-null результат, выигрывает. Найденная культура устанавливается через CultureInfo.CurrentCulture и CultureInfo.CurrentUICulture.

// Пошаговое прохождение запроса через middleware:
// Запрос: GET /api/messages?culture=ru
// Accept-Language: en-US

// Шаг 1: QueryStringRequestCultureProvider
//   Находит ?culture=ru → возвращает ProviderCultureResult("ru")
//   → Проверяем: "ru" есть в SupportedCultures? Да → принимаем

// Шаг 2 (не выполняется): AcceptLanguageHeaderRequestCultureProvider
//   Пропущен — предыдущий провайдер уже вернул результат

// Результат:
//   CultureInfo.CurrentCulture   = "ru-RU" (или "ru" если нет региональной)
//   CultureInfo.CurrentUICulture = "ru"
//   Content-Language: ru (если ApplyCurrentCultureToResponseHeaders = true)

// Теперь во всём запросе:
DateTime.Now.ToString("D")    // → "8 мая 2026 г."
1234.56.ToString("C")         // → "1 234,56 ₽" (если CurrentCulture = "ru-RU")

// Получение текущей культуры в контроллере/сервисе:
public class MessagesController : ControllerBase
{
    [HttpGet("timestamp")]
    public IActionResult GetTimestamp()
    {
        var culture = CultureInfo.CurrentCulture;
        var uiCulture = CultureInfo.CurrentUICulture;

        return Ok(new
        {
            currentCulture   = culture.Name,          // "ru-RU"
            currentUICulture = uiCulture.Name,        // "ru"
            formattedDate    = DateTime.UtcNow.ToString("D", culture), // локализованный формат
            formattedNumber  = 1234.56.ToString("N2", culture)         // "1 234,56"
        });
    }
}

Что это. IStringLocalizer<T> — сервис для получения переведённых строк. Ключ — исходная строка на базовом языке; значение — перевод для текущей культуры. Строки хранятся в .resx файлах.

Структура .resx файлов

// Структура Resources/ папки:

// IStringLocalizer<T> — T определяет имя .resx файла
// Для MessagesController → ищет Resources/Controllers/MessagesController.{culture}.resx

[ApiController]
[Route("api/messages")]
public class MessagesController(
    IChatService chatService,
    IStringLocalizer<MessagesController> localizer) : ControllerBase
{
    [HttpGet("{messageId:guid}")]
    public async Task<IActionResult> GetMessage(Guid messageId)
    {
        var message = await chatService.GetMessageByIdAsync(messageId);

        if (message is null)
        {
            // localizer["ключ"] → LocalizedString
            // Если перевод для текущей культуры не найден → возвращает ключ как есть
            var errorMessage = localizer["MessageNotFound", messageId];
            // Для ru: "Сообщение с ID 3fa85f64... не найдено."
            // Для en: "Message with ID 3fa85f64... was not found."
            // Для неподдерживаемой культуры: "MessageNotFound" (ключ как fallback)

            return NotFound(new ProblemDetails
            {
                Status = 404,
                Title  = localizer["NotFound"].Value,
                Detail = errorMessage.Value
            });
        }

        return Ok(message);
    }

    [HttpPost]
    public async Task<IActionResult> SendMessage([FromBody] SendMessageRequest request)
    {
        var result = await chatService.SendMessageAsync(request);

        // LocalizedString.ResourceNotFound = true если перевод не найден
        var successMsg = localizer["MessageSent"];
        if (successMsg.ResourceNotFound)
            Log.Warning("Missing localization key: {Key}", "MessageSent");

        return Ok(new { message = successMsg.Value, data = result });
    }
}

8.5 Globalization: форматирование без перевода

Зачем отдельно. Иногда нужно только правильное форматирование чисел и дат (без перевода строк). Например, API возвращает валюту в формате пользователя, но сообщения об ошибках — всегда на английском.

// Пример: endpoint возвращает данные в формате культуры пользователя
[HttpGet("balance")]
public IActionResult GetBalance()
{
    var balance = 12345.67m;
    var culture  = CultureInfo.CurrentCulture;

    return Ok(new
    {
        // Форматирование по культуре запроса:
        raw              = balance,                           // 12345.67 (для JSON, всегда)
        formatted        = balance.ToString("N2", culture),  // "12,345.67" (en) / "12 345,67" (ru)
        currency         = balance.ToString("C2", culture),  // "$12,345.67" (en-US) / "12 345,67 ₽" (ru-RU)
        date             = DateTime.UtcNow.ToString("D", culture), // "Thursday, May 8, 2026" / "8 мая 2026 г."
        // Рекомендация для API: всегда возвращать raw + сформатированное отдельно,
        // чтобы клиент мог форматировать сам при необходимости
    });
}

9. Application Monitoring and Observability

Observability — способность понять, что происходит внутри приложения по его внешним сигналам. Три столпа: Logs (события), Metrics (числовые измерения), Traces (цепочки вызовов). Этот раздел охватывает Health Checks и метрики — два наиболее практичных инструмента для Web API.

9.1 Health Checks — AddHealthChecks

Что это. Health Check — механизм для проверки работоспособности приложения и его зависимостей. Kubernetes, Docker Swarm, load balancer'ы используют этот endpoint чтобы решить: направлять ли трафик на экземпляр приложения (liveness) и готово ли оно принимать запросы (readiness).

Базовая настройка

// Program.cs
builder.Services.AddHealthChecks()
    // Проверка подключения к БД — нет отдельного пакета для EF Core/MySQL,
    // проверяем напрямую через DbContext:
    .AddDbContextCheck<ApplicationDbContext>(
        name: "database",
        failureStatus: HealthStatus.Unhealthy,
        tags: ["db", "ready"])
    // ↑ Требует: dotnet add package Microsoft.Extensions.Diagnostics.HealthChecks.EntityFrameworkCore
    //   Выполняет: dbContext.Database.CanConnectAsync()

    // Проверка внешнего HTTP-сервиса:
    .AddUrlGroup(
        uri: new Uri("https://api.dropboxapi.com/2/users/get_current_account"),
        name: "dropbox-api",
        failureStatus: HealthStatus.Degraded, // Degraded — работаем, но с ограничениями
        tags: ["external"])
    // ↑ Требует: dotnet add package AspNetCore.HealthChecks.Uris

    // Проверка Redis (если используется):
    .AddRedis(
        redisConnectionString: builder.Configuration.GetConnectionString("Redis")!,
        name: "redis",
        failureStatus: HealthStatus.Degraded,
        tags: ["cache", "ready"])
    // ↑ Требует: dotnet add package AspNetCore.HealthChecks.Redis

    // Простая проверка доступности памяти:
    .AddCheck("memory", () =>
    {
        var allocated = GC.GetTotalMemory(forceFullCollection: false);
        var threshold = 1024L * 1024 * 512; // 512 MB
        return allocated < threshold
            ? HealthCheckResult.Healthy($"Memory: {allocated / 1024 / 1024} MB")
            : HealthCheckResult.Degraded($"High memory: {allocated / 1024 / 1024} MB");
    }, tags: ["ready"]);

// Маппинг endpoints:
app.MapHealthChecks("/health", new HealthCheckOptions
{
    // Возвращать развёрнутый JSON-ответ вместо простого "Healthy"
    ResponseWriter = UIResponseWriter.WriteHealthCheckUIResponse,
    // ↑ Требует: dotnet add package AspNetCore.HealthChecks.UI.Client
    Predicate = _ => true // все проверки
});

// Liveness endpoint — только критичные проверки (без внешних зависимостей):
app.MapHealthChecks("/health/live", new HealthCheckOptions
{
    Predicate = check => !check.Tags.Contains("external") && !check.Tags.Contains("db")
    // Liveness: приложение не зависло? Не зависит от БД или внешних сервисов.
});

// Readiness endpoint — всё, что нужно для принятия запросов:
app.MapHealthChecks("/health/ready", new HealthCheckOptions
{
    Predicate = check => check.Tags.Contains("ready")
    // Readiness: БД доступна? Redis доступен? Только тогда принимаем трафик.
});

// Ответ /health при WriteHealthCheckUIResponse:

9.2 Custom Health Checks — IHealthCheck

Что это. IHealthCheck — интерфейс для создания произвольных проверок. Любая логика: проверка дискового пространства, доступность очереди сообщений, количество ошибок за последние 5 минут, лицензии и т.д.

// Кастомная проверка: доступность Dropbox API (актуально для WebMessenger)
public class DropboxHealthCheck(
    IFileStorage fileStorage,
    ILogger<DropboxHealthCheck> logger) : IHealthCheck
{
    public async Task<HealthCheckResult> CheckHealthAsync(
        HealthCheckContext context,
        CancellationToken cancellationToken = default)
    {
        try
        {
            // Проверяем доступность хранилища лёгким запросом
            var isAvailable = await fileStorage.PingAsync(cancellationToken);

            if (isAvailable)
                return HealthCheckResult.Healthy("Dropbox storage is reachable.");

            // Degraded: работаем, но аватары загружаться не будут
            return HealthCheckResult.Degraded(
                "Dropbox storage ping failed. Avatar uploads will be unavailable.",
                data: new Dictionary<string, object>
                {
                    ["endpoint"] = "api.dropboxapi.com",
                    ["checkedAt"] = DateTimeOffset.UtcNow
                });
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Dropbox health check failed");

            // Unhealthy: критическая ошибка, описание в exception
            return new HealthCheckResult(
                context.Registration.FailureStatus, // берём из конфигурации
                "Dropbox storage check threw an exception.",
                ex,
                data: new Dictionary<string, object> { ["error"] = ex.Message });
        }
    }
}

// Регистрация кастомной проверки:
builder.Services.AddHealthChecks()
    .AddCheck<DropboxHealthCheck>(
        name: "dropbox-storage",
        failureStatus: HealthStatus.Degraded, // не Unhealthy — приложение работает без аватаров
        tags: ["external", "storage"]);

// Кастомная проверка: мониторинг задержки ответов за последние N минут
public class ResponseTimeHealthCheck(IMetricsService metrics) : IHealthCheck
{
    public Task<HealthCheckResult> CheckHealthAsync(
        HealthCheckContext context,
        CancellationToken cancellationToken = default)
    {
        var avgResponseMs = metrics.GetAverageResponseTimeMs(TimeSpan.FromMinutes(5));

        return Task.FromResult(avgResponseMs switch
        {
            < 200  => HealthCheckResult.Healthy($"Avg response: {avgResponseMs:F0}ms"),
            < 1000 => HealthCheckResult.Degraded($"Slow responses: {avgResponseMs:F0}ms avg"),
            _      => HealthCheckResult.Unhealthy($"Very slow: {avgResponseMs:F0}ms avg")
        });
    }
}

Что это. System.Diagnostics.Metrics — встроенный API .NET 8 для создания кастомных метрик. Он совместим с OpenTelemetry и Prometheus без дополнительных абстракций.

Типы инструментов

UpDownCounter UpDownCounter<T> Счётчик, который может уменьшаться Активные WebSocket соединения, пользователи онлайн

Реализация кастомных метрик для WebMessenger

// Отдельный класс — реестр метрик приложения
public class WebMessengerMetrics : IDisposable
{
    private readonly Meter _meter;

    // Counter: количество отправленных сообщений
    public readonly Counter<long> MessagesSent;

    // Counter: количество входов в систему (с тегами: success/failure)
    public readonly Counter<long> LoginAttempts;

    // UpDownCounter: активные SignalR соединения
    public readonly UpDownCounter<int> ActiveConnections;

    // Histogram: длина сообщений в символах
    public readonly Histogram<int> MessageLength;

    public WebMessengerMetrics(IMeterFactory meterFactory)
    {
        // IMeterFactory — DI-friendly способ создать Meter.
        // Meter будет автоматически включён в OpenTelemetry пайплайн.
        _meter = meterFactory.Create("WebMessenger.Api");

        MessagesSent = _meter.CreateCounter<long>(
            name: "webmessenger.messages.sent",
            unit: "{message}",
            description: "Total number of messages sent through the API");

        LoginAttempts = _meter.CreateCounter<long>(
            name: "webmessenger.auth.login_attempts",
            unit: "{attempt}",
            description: "Total login attempts");

        ActiveConnections = _meter.CreateUpDownCounter<int>(
            name: "webmessenger.signalr.active_connections",
            unit: "{connection}",
            description: "Number of active SignalR connections");

        MessageLength = _meter.CreateHistogram<int>(
            name: "webmessenger.messages.length",
            unit: "{character}",
            description: "Distribution of message lengths in characters");
    }

    public void Dispose() => _meter.Dispose();
}

// Регистрация:
builder.Services.AddSingleton<WebMessengerMetrics>();
// Singleton — метрики разделяются всеми запросами

// Использование метрик в сервисах:
public class ChatService(
    IUnitOfWork uow,
    ICurrentUser currentUser,
    WebMessengerMetrics metrics) // ← инжектируем метрики
{
    public async Task<MessageDto> SendMessageAsync(
        Guid recipientId,
        SendMessageRequest request)
    {
        var message = await SaveMessageAsync(request);

        // Записываем метрику с тегами
        metrics.MessagesSent.Add(1,
            new TagList
            {
                { "chat.type", "direct" },        // тег: тип чата
                { "user.id",   currentUser.Id }   // тег: ID отправителя
            });

        // Histogram: длина сообщения
        metrics.MessageLength.Record(
            request.Content.Length,
            new TagList { { "chat.type", "direct" } });

        return message.ToDto();
    }
}

ObservableGauge — метрики по запросу

// ObservableGauge вычисляется в момент скрапинга Prometheus,
// а не при каждом запросе — подходит для "тяжёлых" метрик.

public class WebMessengerMetrics : IDisposable
{
    private readonly Meter _meter;
    private readonly IServiceScopeFactory _scopeFactory;

    public WebMessengerMetrics(IMeterFactory meterFactory, IServiceScopeFactory scopeFactory)
    {
        _meter        = meterFactory.Create("WebMessenger.Api");
        _scopeFactory = scopeFactory;

        // ObservableGauge: количество зарегистрированных пользователей
        _meter.CreateObservableGauge<long>(
            name: "webmessenger.users.total",
            observeValue: GetTotalUsersCount,
            unit: "{user}",
            description: "Total number of registered users");

        // ObservableGauge: размер папки логов
        _meter.CreateObservableGauge<long>(
            name: "webmessenger.logs.size_bytes",
            observeValue: GetLogsFolderSize,
            unit: "By");
    }

    private long GetTotalUsersCount()
    {
        // Используем новый scope для доступа к Scoped-сервисам из Singleton
        using var scope = _scopeFactory.CreateScope();
        var uow = scope.ServiceProvider.GetRequiredService<IUnitOfWork>();
        return uow.UserRepository.GetAll().LongCount();
        // Внимание: синхронный вызов LongCount() при каждом скрапинге.
        // Для дорогих запросов — кэшировать результат в поле с периодическим обновлением.
    }

    private long GetLogsFolderSize()
    {
        var logsPath = Path.Combine(AppContext.BaseDirectory, "logs");
        if (!Directory.Exists(logsPath)) return 0;
        return new DirectoryInfo(logsPath)
            .GetFiles("*.txt")
            .Sum(f => f.Length);
    }

    public void Dispose() => _meter.Dispose();
}
